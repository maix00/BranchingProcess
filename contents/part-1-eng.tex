\section{\bilingual{Lemmas on Applying \texorpdfstring{Mogul'ski\u i}{Mogul'skii}'s Theorem}{有关应用 \texorpdfstring{Mogul'ski\u i}{Mogul'skii} 定理的引理}}\label{section:mogulskii-lemmas}

\bilingual{
    In this section, $\Delta>0$ denotes the tube width, and $L:(0,\infty)\to\mathbb Z_{\ge0}$ assigns to each width $\Delta$ its observation time $L_\Delta$, measured in random-walk steps. We assume that this time horizon is asymptotically larger than the diffusive scale $\Delta^2$, in the precise sense
}{
    本节中, $\Delta>0$ 表示管道宽度, $L:(0,\infty)\to\mathbb Z_{\ge0}$ 将每个宽度 $\Delta$ 对应到观测时间 $L_\Delta$, 即随机游走步数. 我们假设该时间长度渐近大于扩散尺度 $\Delta^2$, 精确地说
}
\begin{equation}
    \lim_{\Delta\to\infty}L_\Delta=\infty,
    \qquad
    \lim_{\Delta\to\infty}\frac{L_\Delta}{\Delta^2}=\infty.
    \label{eq:mogulskii-L-delta-assumptions}
\end{equation}
\begin{lemma}
    \label{lemma:mogulskii-cal}
    \bilingual{
        Consider the zero-mean random walk $(Y_n)_{n\ge0}$ with finite variance $0<\sigma^2<\infty$ obtained from \nameref{lemma:many-to-one}. Let $(\xi_i)_{i\ge1}$ be its standardized increments, so that
    }{
        考虑通过 \nameref{lemma:many-to-one} 得到的零均值和有限方差 $0<\sigma^2<\infty$ 的随机游走 $(Y_n)_{n\ge0}$. 令 $(\xi_i)_{i\ge1}$ 为其标准化增量, 即
    }
    \begin{equation*}
        \xi_i:=\frac{Y_i-Y_{i-1}}{\sigma},\qquad
        Y_j=\sigma\sum_{i=1}^{j}\xi_i,\qquad j\in\mathbb Z_{\ge0}.
    \end{equation*}
    \bilingual{
        Then $(\xi_i)_{i\ge1}$ are IID, centered, and have unit variance. Fix $a\in(0,1)$. The scale assumption is \eqref{eq:mogulskii-L-delta-assumptions}.
    }{
        则 $(\xi_i)_{i\ge1}$ 独立同分布, 均值为 $0$, 方差为 $1$. 固定 $a\in(0,1)$, 并假设尺度条件 \eqref{eq:mogulskii-L-delta-assumptions} 成立.
    }
    \begin{enumerate}[label=\upshape(\roman*)]
        \item \bilingual{Define}{定义}
            \begin{equation*}
                \mathcal T_\Delta:=
                \bigcap_{j=1}^{L_\Delta}
                    \left\{-a\Delta\le Y_j\le(1-a)\Delta\right\}.
            \end{equation*}
            \bilingual{Then}{则}
            \begin{equation*}
                \lim_{\Delta\to\infty}\frac{\Delta^2}{L_\Delta}
                    \log\mathbb P(\mathcal T_\Delta)
                =-\frac{\pi^2}{2}\sigma^2.
            \end{equation*}
        \item \bilingual{
            Fix $0<u<v<1$. Let $v_1,v_2:(0,\infty)\to\mathbb R$ satisfy
        }{
            固定 $0<u<v<1$. 设 $v_1,v_2:(0,\infty)\to\mathbb R$ 满足
        }
            \begin{equation*}
                \exists \Delta_0>0,\
                \forall\Delta\ge\Delta_0:\quad
                v_1(\Delta)\le u<v\le v_2(\Delta).
            \end{equation*}
            \bilingual{Define}{定义}
            \begin{equation*}
                \mathcal E_\Delta:=
                    \mathcal T_\Delta\cap
                    \left\{(v_1(\Delta)-a)\Delta\le Y_{L_\Delta}
                        \le(v_2(\Delta)-a)\Delta\right\}.
            \end{equation*}
            \bilingual{Then}{则}
            \begin{equation*}
                \lim_{\Delta\to\infty}\frac{\Delta^2}{L_\Delta}
                    \log\mathbb P(\mathcal E_\Delta)
                =-\frac{\pi^2}{2}\sigma^2.
            \end{equation*}
    \end{enumerate}
\end{lemma}
\begin{proof}
    \bilingual{
        Use the convention $\log0=-\infty$. The applications of
        \nameref{theorem:mogulskii} below give positive event probabilities
        eventually along each selected subsequence.
    }{
        对零概率事件约定 $\log0=-\infty$. 下文应用
        \nameref{theorem:mogulskii} 将给出所取子列上事件概率最终为正.
    }
    \bilingual{
        By (3.1), choose $\Delta_*>0$ such that $L_\Delta\ge1$ for all
        $\Delta\ge\Delta_*$. The rate functions below are defined on this tail.
    }{
        由 (3.1), 取 $\Delta_*>0$ 使得对所有 $\Delta\ge\Delta_*$ 都有
        $L_\Delta\ge1$. 下文的速率函数均定义在此尾区间上.
    }
    \begin{enumerate}[label=(\roman*)]
        \item
            \bilingual{
            Set $c=-\frac{\pi^2}{2}\sigma^2$ and
            $\rho_{\mathrm{tube}}(\Delta)=\frac{\Delta^2}{L_\Delta}
                \log\mathbb P(\mathcal T_\Delta)$. For $\eta>0$, define
            }{
            记 $c=-\frac{\pi^2}{2}\sigma^2$ 及
            $\rho_{\mathrm{tube}}(\Delta)=\frac{\Delta^2}{L_\Delta}
                \log\mathbb P(\mathcal T_\Delta)$. 对 $\eta>0$, 定义
            }
            \begin{equation*}
                \mathcal B_{\mathrm{tube},\eta}:=
                \left\{\Delta\in[\Delta_*,\infty):
                    \eta\le|\rho_{\mathrm{tube}}(\Delta)-c|\right\}.
            \end{equation*}
            \bilingual{
            The definition of the limit reduces the claim to boundedness of
            $\mathcal B_{\mathrm{tube},\eta}$. If this set is unbounded, choose
            $1\le\Delta_1<\Delta_2<\cdots$ in $\mathcal B_{\mathrm{tube},\eta}$,
            and set $L_k:=L_{\Delta_k}$ and $n_k:=L_k+1$.
            }{
            由极限定义, 只需证明 $\mathcal B_{\mathrm{tube},\eta}$ 有界. 若该集合无界,
            取其中严格递增的数列 $1\le\Delta_1<\Delta_2<\cdots$, 并记
            $L_k:=L_{\Delta_k}$ 和 $n_k:=L_k+1$.
            }
            \bilingual{
            By \eqref{eq:mogulskii-L-delta-assumptions}, $L_k/\Delta_k^2\to\infty$ and hence $L_k\to\infty$. Pass to a further subsequence with $L_k$ strictly increasing, and define
            }{
            由 \eqref{eq:mogulskii-L-delta-assumptions}, $L_k/\Delta_k^2\to\infty$, 从而 $L_k\to\infty$. 取进一步子列使 $L_k$ 严格递增, 并定义
            }
            \begin{equation*}
            \widehat{x}:\mathbb Z_{\ge0}\to(0,\infty),\qquad
            \widehat{x}(n):=
                \begin{cases}
                    1,&0\le n<n_1,\\
                    \Delta_k,&n_k\le n<n_{k+1}.
                \end{cases}
            \end{equation*}
            \bilingual{
                For $n_k\le n<n_{k+1}$, we have
                $\widehat{x}(n)=\Delta_k\to\infty$ and
                $0\le\frac{\widehat{x}(n)^2}{n}
                    \le\frac{\Delta_k^2}{L_k+1}\to0$.
            }{
                当 $n_k\le n<n_{k+1}$ 时,
                $\widehat{x}(n)=\Delta_k\to\infty$, 且
                $0\le\frac{\widehat{x}(n)^2}{n}
                    \le\frac{\Delta_k^2}{L_k+1}\to0$.
            }
            \bilingual{
                Use $s_n$ from
                \eqref{eq:mogulskii-s-n-t} with $x(n)=\widehat{x}(n)$.
            }{
                在
                \eqref{eq:mogulskii-s-n-t} 中取 $x(n)=\widehat{x}(n)$.
            }
            \begin{equation*}
                s_{n_k}(t)=\frac{Y_j}{\sigma\Delta_k}
                \quad\left(\frac{j}{n_k}\le t<\frac{j+1}{n_k},\ 0\le j\le L_k\right),
                \qquad
                s_{n_k}(1)=\frac{Y_{L_k}}{\sigma\Delta_k}.
            \end{equation*}
            \bilingual{
                For $0<\varepsilon<\frac{\min\{a,1-a\}}{\sigma}$, define
            }{
                对 $0<\varepsilon<\frac{\min\{a,1-a\}}{\sigma}$, 定义
            }
            \begin{align*}
                S_2^-(t)&:=-\frac{a}{\sigma}+\varepsilon,
                &S_1^-(t)&:=\frac{1-a}{\sigma}-\varepsilon,\\
                S_2^+(t)&:=-\frac{a}{\sigma}-\varepsilon,
                &S_1^+(t)&:=\frac{1-a}{\sigma}+\varepsilon.
            \end{align*}
            \bilingual{
                Let $G^-_\varepsilon$ and $G^+_\varepsilon$ be the sets in
                \eqref{eq:mogulskii-G} for the boundary pairs
                $(S_1^-,S_2^-)$ and $(S_1^+,S_2^+)$, respectively, and put
                $C:=\left\{f\in D(0,1):f(0)=0,\ f([0,1])\subseteq
                \left[-\frac{a}{\sigma},\frac{1-a}{\sigma}\right]\right\}$.
            }{
                令 $G^-_\varepsilon$ 和 $G^+_\varepsilon$ 分别为
                \eqref{eq:mogulskii-G} 中对应边界对
                $(S_1^-,S_2^-)$ 和 $(S_1^+,S_2^+)$ 的集合, 并令
                $C:=\left\{f\in D(0,1):f(0)=0,\ f([0,1])\subseteq
                \left[-\frac{a}{\sigma},\frac{1-a}{\sigma}\right]\right\}$.
            }
            \bilingual{
                the corridors $G^-_\varepsilon,G^+_\varepsilon$ satisfy
                \eqref{eq:mogulskii-S-1-S-2} and contain the initial value $0$.
                The inclusion $G^-_\varepsilon\subset C\subset G^+_\varepsilon$ holds,
                and their boundary functions satisfy
                $S_2^\pm(0)<0<S_1^\pm(0)$ and
                $S_2^\pm(t)<S_1^\pm(t)$ for $t\in[0,1]$.
            }{
                管道 $G^-_\varepsilon,G^+_\varepsilon$ 满足
                \eqref{eq:mogulskii-S-1-S-2}, 且包含初值 $0$.
                包含关系为 $G^-_\varepsilon\subset C\subset G^+_\varepsilon$,
                且边界函数满足 $S_2^\pm(0)<0<S_1^\pm(0)$ 及
                $S_2^\pm(t)<S_1^\pm(t)$ ($t\in[0,1]$).
            }
            \bilingual{
                Their widths are $S_1^\pm-S_2^\pm=\frac{1}{\sigma}\mp2\varepsilon$,
                so, with matching signs,
                $H_2(G^\pm_\varepsilon)
                =\int_0^1\left(\frac{1}{\sigma}\mp2\varepsilon\right)^{-2}\,dt
                =\left(\frac{1}{\sigma}\mp2\varepsilon\right)^{-2}$.
            }{
                其宽度为 $S_1^\pm-S_2^\pm=\frac{1}{\sigma}\mp2\varepsilon$,
                因而按对应符号有
                $H_2(G^\pm_\varepsilon)
                =\int_0^1\left(\frac{1}{\sigma}\mp2\varepsilon\right)^{-2}\,dt
                =\left(\frac{1}{\sigma}\mp2\varepsilon\right)^{-2}$.
            }
            \bilingual{
                At $n=n_k$, $\widehat{x}(n_k)=\Delta_k$. Define the tube event
            }{
                当 $n=n_k$ 时, $\widehat{x}(n_k)=\Delta_k$. 定义管道事件
            }
            \begin{equation*}
                \mathcal T_k:=
                \bigcap_{j=1}^{L_k}
                \left\{-a\Delta_k\le Y_j\le(1-a)\Delta_k\right\},
                \qquad
                \{s_{n_k}\in C\}=\mathcal T_k.
            \end{equation*}
            \bilingual{
                Put $q_k:=\frac{\Delta_k^2}{n_k}\log\mathbb P(s_{n_k}\in C)$.
                Applying \nameref{theorem:mogulskii} to $G^-_\varepsilon$ and
                $G^+_\varepsilon$ along $\widehat{x}$ gives
            }{
                记 $q_k:=\frac{\Delta_k^2}{n_k}\log\mathbb P(s_{n_k}\in C)$.
                沿 $\widehat{x}$ 对 $G^-_\varepsilon$ 与 $G^+_\varepsilon$ 应用
                \nameref{theorem:mogulskii}, 得
            }
            \begin{equation*}
                -\frac{\pi^2}{2}\left(\frac{1}{\sigma}-2\varepsilon\right)^{-2}
                \le\liminf_{k\to\infty}q_k
                \le\limsup_{k\to\infty}q_k
                \le-\frac{\pi^2}{2}\left(\frac{1}{\sigma}+2\varepsilon\right)^{-2}.
            \end{equation*}
            \bilingual{
                The finite Mogulskii rate for $G^-_\varepsilon$ gives
                $\mathbb P(s_{n_k}\in G^-_\varepsilon)>0$ eventually; since
                $G^-_\varepsilon\subset C$ and $\{s_{n_k}\in C\}=\mathcal T_k$,
                $q_k$ is real for all sufficiently large $k$.
            }{
                $G^-_\varepsilon$ 的 Mogulskii 极限有限, 故充分大的 $k$ 满足
                $\mathbb P(s_{n_k}\in G^-_\varepsilon)>0$. 由
                $G^-_\varepsilon\subset C$ 及 $\{s_{n_k}\in C\}=\mathcal T_k$,
                $q_k$ 对充分大的 $k$ 为实数.
            }
            \bilingual{
            Letting $\varepsilon\downarrow0$ in the bound above and using
            $\frac{n_k}{L_k}=1+\frac{1}{L_k}\to1$ gives
            }{
            令 $\varepsilon\downarrow0$ 并利用
            $\frac{n_k}{L_k}=1+\frac{1}{L_k}\to1$, 得
            }
            \begin{align*}
                q_k&\longrightarrow c,\\
                \frac{\Delta_k^2}{L_k}\log\mathbb P(\mathcal T_k)
                &=\frac{n_k}{L_k}q_k\longrightarrow c.
            \end{align*}
            \bilingual{
            The limits above give
            $\eta\le|\rho_{\mathrm{tube}}(\Delta_k)-c|\to0$, which forces $\eta=0$.
            Thus $\mathcal B_{\mathrm{tube},\eta}$ is bounded. Since $\eta>0$ was arbitrary,
            the definition of the limit gives $\lim_{\Delta\to\infty}\rho_{\mathrm{tube}}(\Delta)=c$.
            }{
            上述极限给出
            $\eta\le|\rho_{\mathrm{tube}}(\Delta_k)-c|\to0$, 故 $\eta=0$.
            因而 $\mathcal B_{\mathrm{tube},\eta}$ 有界. 由于 $\eta>0$ 任意,
            由极限定义得到 $\lim_{\Delta\to\infty}\rho_{\mathrm{tube}}(\Delta)=c$.
            }
        \item
            \bilingual{
            Set $c=-\frac{\pi^2}{2}\sigma^2$ and
            $\rho_{\mathrm{endpoint}}(\Delta)=\frac{\Delta^2}{L_\Delta}
                \log\mathbb P(\mathcal E_\Delta)$. For $\eta>0$, define
            }{
            记 $c=-\frac{\pi^2}{2}\sigma^2$ 及
            $\rho_{\mathrm{endpoint}}(\Delta)=\frac{\Delta^2}{L_\Delta}
                \log\mathbb P(\mathcal E_\Delta)$. 对 $\eta>0$, 定义
            }
            \begin{equation*}
                \mathcal B_{\mathrm{endpoint},\eta}:=
                \left\{\Delta\in[\Delta_*,\infty):
                    \eta\le|\rho_{\mathrm{endpoint}}(\Delta)-c|\right\}.
            \end{equation*}
            \bilingual{
            The definition of the limit reduces the claim to boundedness of
            $\mathcal B_{\mathrm{endpoint},\eta}$. If this set is unbounded,
            choose a strictly increasing sequence
            $1\le\Delta_1<\Delta_2<\cdots$ in this set, and set
            $L_k:=L_{\Delta_k}$ and $n_k:=L_k+1$.
            }{
            由极限定义, 只需证明 $\mathcal B_{\mathrm{endpoint},\eta}$ 有界.
            若该集合无界, 取其中严格递增的数列
            $1\le\Delta_1<\Delta_2<\cdots$, 并令
            $L_k:=L_{\Delta_k}$ 和 $n_k:=L_k+1$.
            }
            \bilingual{
            By \eqref{eq:mogulskii-L-delta-assumptions}, $L_k\to\infty$. Passing to a further subsequence, assume that $L_k$ is strictly increasing, and set
            }{
            由 \eqref{eq:mogulskii-L-delta-assumptions} 可知 $L_k\to\infty$. 再取进一步子列使 $L_k$ 严格递增, 并令
            }
            \begin{equation*}
            \widehat{x}:\mathbb Z_{\ge0}\to(0,\infty),\qquad
            \widehat{x}(n):=
                \begin{cases}
                    1,&0\le n<n_1,\\
                    \Delta_k,&n_k\le n<n_{k+1}.
                \end{cases}
            \end{equation*}
            \bilingual{
            For $n_k\le n<n_{k+1}$, we have
            $\widehat{x}(n)=\Delta_k\to\infty$ and
            $0\le\frac{\widehat{x}(n)^2}{n}
                \le\frac{\Delta_k^2}{L_k+1}\to0$.
            }{
            当 $n_k\le n<n_{k+1}$ 时,
            $\widehat{x}(n)=\Delta_k\to\infty$, 且
            $0\le\frac{\widehat{x}(n)^2}{n}
                \le\frac{\Delta_k^2}{L_k+1}\to0$.
            }
            \bilingual{
            Let $s_n$ be given by \eqref{eq:mogulskii-s-n-t}.
            }{
            令 $s_n$ 由 \eqref{eq:mogulskii-s-n-t} 定义.
            }
            \bilingual{
                Define the boundaries and the path class
            }{
                定义边界函数及路径类
            }
            \begin{equation*}
                \begin{alignedat}{2}
                    S_2(t)&=
                    \begin{cases}
                        -\frac{a}{\sigma},&0\le t<1,\\
                        \frac{u-a}{\sigma},&t=1,
                    \end{cases}
                    \qquad&
                    S_1(t)&=
                    \begin{cases}
                        \frac{1-a}{\sigma},&0\le t<1,\\
                        \frac{v-a}{\sigma},&t=1.
                    \end{cases}
                \end{alignedat}
            \end{equation*}
            \begin{equation*}
                G:=\{f\in D(0,1):f(0)=0,\
                    S_2(t)<f(t)<S_1(t)\ \text{for all }t\in[0,1]\}.
            \end{equation*}
            \bilingual{
                The boundary condition in \eqref{eq:mogulskii-S-1-S-2} follows from
                $\max\{-a,u-a\}=u-a<v-a=\min\{1-a,v-a\}$ and $-a<0<1-a$.
                The energy is $H_2(G)=\int_0^1(S_1(t)-S_2(t))^{-2}\,dt
                =\int_0^1\sigma^2\,dt=\sigma^2$.
            }{
                条件 \eqref{eq:mogulskii-S-1-S-2} 由
                $\max\{-a,u-a\}=u-a<v-a=\min\{1-a,v-a\}$ 和 $-a<0<1-a$ 成立.
                其能量为 $H_2(G)=\int_0^1(S_1(t)-S_2(t))^{-2}\,dt
                =\int_0^1\sigma^2\,dt=\sigma^2$.
            }
            \bilingual{
                For $0\le j\le L_k$ and at the terminal time,
            }{
                对 $0\le j\le L_k$ 及终点时刻,
            }
            \begin{equation*}
                \begin{aligned}
                    s_{n_k}(t)&=\frac{Y_j}{\sigma\Delta_k}
                    \quad\left(\frac{j}{n_k}\le t<\frac{j+1}{n_k},\ 0\le j\le L_k\right),\\
                    s_{n_k}(1)&=\frac{Y_{L_k}}{\sigma\Delta_k}.
                \end{aligned}
            \end{equation*}
            \bilingual{
                Write $\mathcal G_k:=\{s_{n_k}\in G\}$. Its exact event identity is
            }{
                记 $\mathcal G_k:=\{s_{n_k}\in G\}$. 则有以下事件恒等式
            }
            \begin{equation*}
                \begin{aligned}
                    \mathcal G_k
                    &=
                    \left(\bigcap_{j=1}^{L_k}
                        \left\{-a\Delta_k<Y_j<(1-a)\Delta_k\right\}\right)\\
                    &\quad\cap
                    \left\{(u-a)\Delta_k<Y_{L_k}<(v-a)\Delta_k\right\}.
                \end{aligned}
            \end{equation*}
            \bilingual{
                Also, $S_1(t)-S_2(t)=\frac{1}{\sigma}$ for $0\le t<1$, and the endpoint assumption gives $v_1(\Delta_k)\le u<v\le v_2(\Delta_k)$ for all sufficiently large $k$. Thus
            }{
                此外, $0\le t<1$ 时 $S_1(t)-S_2(t)=\frac{1}{\sigma}$, 且端点条件给出充分大的 $k$ 满足 $v_1(\Delta_k)\le u<v\le v_2(\Delta_k)$. 因而
            }
            \begin{align*}
                \mathcal T_k
                &:=\bigcap_{j=1}^{L_k}
                    \left\{-a\Delta_k\le Y_j\le(1-a)\Delta_k\right\}
                    =\mathcal T_{\Delta_k},\\
                \mathcal E_k
                &:=\mathcal T_k\cap
                    \{(v_1(\Delta_k)-a)\Delta_k\le Y_{L_k}
                        \le(v_2(\Delta_k)-a)\Delta_k\}
                    =\mathcal E_{\Delta_k}.
            \end{align*}
            \begin{equation*}
                (v_1(\Delta_k)-a)\Delta_k
                \le(u-a)\Delta_k<Y_{L_k}
                <(v-a)\Delta_k\le(v_2(\Delta_k)-a)\Delta_k.
            \end{equation*}
            \bilingual{
                Thus $\mathcal G_k\subseteq\mathcal E_k\subseteq\mathcal T_k$.
            }{
                因而 $\mathcal G_k\subseteq\mathcal E_k\subseteq\mathcal T_k$.
            }
            \bilingual{
                By \nameref{theorem:mogulskii}, set
                $q_k:=\frac{\Delta_k^2}{n_k}\log\mathbb P(\mathcal G_k)$.
                Then $q_k\to-\frac{\pi^2}{2}H_2(G)=c$ since $H_2(G)=\sigma^2$.
                Hence $\mathbb P(\mathcal G_k)>0$ eventually, and
                $\mathcal G_k\subseteq\mathcal E_k\subseteq\mathcal T_k$
                makes their probabilities positive as well.
                Also,
                $\frac{n_k}{L_k}=1+\frac{1}{L_k}\to1$,
            }{
                由 \nameref{theorem:mogulskii}, 定义
                $q_k:=\frac{\Delta_k^2}{n_k}\log\mathbb P(\mathcal G_k)$.
                结合 $H_2(G)=\sigma^2$, 有 $q_k\to-\frac{\pi^2}{2}H_2(G)=c$.
                因而充分大的 $k$ 满足 $\mathbb P(\mathcal G_k)>0$;
                结合 $\mathcal G_k\subseteq\mathcal E_k\subseteq\mathcal T_k$,
                后两事件的概率也为正.
                又因
                $\frac{n_k}{L_k}=1+\frac{1}{L_k}\to1$,
            }
            \begin{equation*}
                \frac{\Delta_k^2}{L_k}\log\mathbb P(\mathcal G_k)
                =\frac{n_k}{L_k}q_k\longrightarrow c.
            \end{equation*}
            \bilingual{
                Applying (i) to $(L_\Delta)$ gives
                $\frac{\Delta_k^2}{L_k}\log\mathbb P(\mathcal T_k)\to c$.
            }{
                对序列 $(L_\Delta)$ 应用 (i), 得
                $\frac{\Delta_k^2}{L_k}\log\mathbb P(\mathcal T_k)\to c$.
            }
            \bilingual{
                For an event $A$, set
                $Q_k(A):=\frac{\Delta_k^2}{L_k}\log\mathbb P(A)$.
                Since $\Delta_k^2/L_k>0$, the event inclusion gives
            }{
                对事件 $A$, 记
                $Q_k(A):=\frac{\Delta_k^2}{L_k}\log\mathbb P(A)$.
                由于 $\Delta_k^2/L_k>0$, 事件包含关系给出
            }
            \begin{align*}
                c=\lim_{k\to\infty}Q_k(\mathcal G_k)
                &\le\liminf_{k\to\infty}Q_k(\mathcal E_k)
                    \le\limsup_{k\to\infty}Q_k(\mathcal E_k)\\
                &\le\lim_{k\to\infty}Q_k(\mathcal T_k)=c.
            \end{align*}
            \bilingual{
            The limits give $\eta\le|\rho_{\mathrm{endpoint}}(\Delta_k)-c|\to0$, which forces $\eta=0$.
            Hence $\mathcal B_{\mathrm{endpoint},\eta}$ is bounded. Since $\eta>0$ was arbitrary,
            $\lim_{\Delta\to\infty}\rho_{\mathrm{endpoint}}(\Delta)=c$.
            }{
            上述极限给出 $\eta\le|\rho_{\mathrm{endpoint}}(\Delta_k)-c|\to0$, 故 $\eta=0$.
            因而 $\mathcal B_{\mathrm{endpoint},\eta}$ 有界. 由于 $\eta>0$ 任意,
            $\lim_{\Delta\to\infty}\rho_{\mathrm{endpoint}}(\Delta)=c$.
            }
        \qedhere
    \end{enumerate}
\end{proof}
