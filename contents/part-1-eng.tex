\section{\bilingual{Lemmas on Applying \texorpdfstring{Mogul'ski\u i}{Mogul'skii}'s Theorem}{有关应用 \texorpdfstring{Mogul'ski\u i}{Mogul'skii} 定理的引理}}\label{section:mogulskii-lemmas}

\bilingual{
    In this section,~ we introduce a nonnegative integer parameter $\Delta$ (which will later serve as the tube width in the small deviation estimates of the next section) and an integer-valued sequence $(L_\Delta)_{\Delta\in\mathbb Z_{\ge 0}}$. The main purpose is to reformulate \nameref{theorem:mogulskii} into a form suitable for the tube estimates,~ where the roles of $n$ and $x(n)$ are exchanged via a generalized inverse.
}{
    本节引入一个非负整数参数 $\Delta$ (在下一节中它将作为管道宽度参与小偏差估计) 和一个取非负整数值的序列 $(L_\Delta)_{\Delta\in\mathbb Z_{\ge 0}}$. 主要目标是将 \nameref{theorem:mogulskii}改写为适合管道估计的形式, 其中通过广义逆交换 $n$ 与 $x(n)$ 的角色.
}

\bilingual{
    Now consider a nondecreasing sequence $(L_\Delta)_{\Delta\in\mathbb Z_{\ge 0}}$ with values in $\mathbb Z_{\ge 0}$ such that
}{
    现在考虑一个取非负整数值的非递减序列 $(L_\Delta)_{\Delta\in\mathbb Z_{\ge 0}}$, 满足
}
\begin{equation*}
     \lim_{\Delta\to\infty}L_\Delta=\infty,
     \quad
     \lim_{\Delta\to\infty}\frac{L_\Delta}{\Delta^2}=\infty.
\end{equation*}
\bilingual{
    Basically, when applying \nameref{theorem:mogulskii}, we are letting $n$ there to be $L_\Delta$ and $x(n)$ there to be $\Delta$, to form an inverse sequence. We are in this section to make rigorous this statement. Let
}{
    在应用 \nameref{theorem:mogulskii}时, 要让定理中的 $n$ 等于 $L_\Delta$, 定理中的 $x(n)$ 等于 $\Delta$, 从而形成一个反函数序列. 本节将严格论证这一点. 令
}
\begin{align*}
    x(n)&:=\min\{\Delta\in\mathbb Z_{\ge 0}: L_\Delta+1\ge n\},&n\in\mathbb Z_{\ge 0},
    \\ y(n)&:= x(n+1)-1,&n\in\mathbb Z_{\ge 0}.
\end{align*}

\begin{lemma}\label{lemma:mogulskii-x-y}
    \bilingual{
        The two sequences $x$ and $y$ satisfy
    }{
        序列 $x$ 和 $y$ 满足
    }
    $
        \displaystyle\lim_{n\to\infty}x(n)=\lim_{n\to\infty}y(n)=\infty.
    $
    \bilingual{
        Since $\displaystyle\lim_{\Delta\to\infty}\frac{L_\Delta}{\Delta^2}=\infty$, we have
    }{
        由 $\displaystyle\lim_{\Delta\to\infty}\frac{L_\Delta}{\Delta^2}=\infty$ 可知
    }
    {
        \setlength\belowdisplayskip{-15pt}
        \begin{equation*}
            \lim_{n\to\infty}x(n)n^{-1/2}
            =\lim_{n\to\infty}y(n)n^{-1/2}
            =0.
        \end{equation*}
    }
\end{lemma}
\bilingual{
    \noindent The proof for \Cref{lemma:mogulskii-x-y} uses the minimality in the definition of $x$: whenever $x(n)>0$, one has $L_{x(n)-1}+1<n$. Together with $x(n)\to\infty$ and $L_\Delta/\Delta^2\to\infty$, this gives $x^2(n)/n\to0$; the same argument applies to $y$.
}{
    \noindent \Cref{lemma:mogulskii-x-y} 的证明使用$x$定义中的最小性: 当$x(n)>0$时, 有$L_{x(n)-1}+1<n$. 结合$x(n)\to\infty$与$L_\Delta/\Delta^2\to\infty$, 可得$x^2(n)/n\to0$; 对$y$可作相同论证.
}

\bilingual{
    Consider the random walk $(Y_n)_{n\ge 0}$ with zero mean and finite variance $0<\sigma^2<\infty$ obtained from \nameref{lemma:many-to-one}. For any $a\in(0,1)$, we denote in this section
}{
    考虑通过\nameref{lemma:many-to-one}得到的零均值和有限方差 $0<\sigma^2<\infty$ 的随机游走 $(Y_n)_{n\ge 0}$. 对于任意 $a\in(0,1)$, 在本节中定义
}
\begin{align*}
    q(\Delta,n)&:=\log\mathbb P\left(
        -a\Delta\le Y_j\le (1-a)\Delta,\,
        \forall j\in[\![1,n]\!]
    \right),\\
    p(\Delta,n)&:=\frac{\Delta^2}nq(\Delta,n).
\end{align*}
\bilingual{
    Let $n=n(\Delta):=L_\Delta+1$ such that $x(n)\le\Delta<x(n+1)$, and we can get by monotonicity of probability that
}{
    令$n=n(\Delta):=L_\Delta+1$ 使得 $x(n)\le\Delta<x(n+1)$, 并且可由概率单调性得到
}
\begin{lemma} \label{lemma:mogulskii-mono}
    \bilingual{
        For $\Delta_1\le\Delta_2$ such that $L:=L_{\Delta_1}=L_{\Delta_2}$, we have $q(\Delta_1,L)\le q(\Delta_2,L)$.
    }{
        对于满足 $L:=L_{\Delta_1}=L_{\Delta_2}$ 的 $\Delta_1\le\Delta_2$, 有 $q(\Delta_1,L)\le q(\Delta_2,L)$.
    }
\end{lemma}

\bilingual{
    We are now able to get to the main lemma with an additional condition for $(L_\Delta)$.
}{
    现在可在对 $(L_\Delta)$ 的额外条件下得到如下主要引理.
}
\begin{lemma}
    \label{lemma:mogulskii-cal}
    \quad
    \begin{enumerate}[label=\upshape(\roman*)]
        \item \bilingual{
            By \nameref{theorem:mogulskii}, we have, indexed by $n$,
        }{
            根据 \nameref{theorem:mogulskii}, 按 $n$ 索引,
        }
            \begin{equation*}
                \lim_{n\to\infty}p(x(n),n-1)
                =\lim_{n\to\infty}p(y(n),n-1)
                =-\frac{\pi^2}2\sigma^2.
            \end{equation*}
        \bilingual{
            For $n=n(\Delta)$, and indexed by $\Delta$,
        }{
            对于 $n=n(\Delta)$, 按 $\Delta$ 索引,
        }
            \begin{equation*}
                \lim_{\Delta\to\infty}p(x(n),n-1)
                =\lim_{\Delta\to\infty}p(y(n),n-1)
                =-\frac{\pi^2}2\sigma^2.
            \end{equation*}
        \item \bilingual{
            Under the additional condition for $(L_\Delta)$ that, when $\Delta\to\infty$,
        }{
            在 $(L_\Delta)$ 的如下额外条件下, 即当 $\Delta\to\infty$ 时,
        }
            \begin{equation*}
                N_\Delta:=\#\left\{\hat\Delta\ge 0: L_\Delta=L_{\hat\Delta}\right\}
                =O_\Delta(1),
            \end{equation*}
            \bilingual{
                we have
            }{
                成立
            }
            \begin{equation*}
                \lim_{\Delta\to\infty}\frac{\Delta^2}{L_\Delta}
                \log\mathbb P\left(
                    -a\Delta\le Y_j\le (1-a)\Delta,\,
                    \forall j\in[\![1,L_\Delta]\!]
                \right)
                =-\frac{\pi^2}2\sigma^2.
            \end{equation*}
        \item
            \bilingual{
                Under the standing scale condition $L_\Delta/\Delta^2\to\infty$, fix $0<u<v<1$. Let $v_1=v_1(\Delta)$ and $v_2=v_2(\Delta)$ be endpoint-window parameters such that, for all sufficiently large $\Delta$, $v_1(\Delta)\le u$ and $v_2(\Delta)\ge v$. Then
            }{
                在已有尺度条件 $L_\Delta/\Delta^2\to\infty$ 下, 固定 $0<u<v<1$. 设 $v_1=v_1(\Delta)$ 与 $v_2=v_2(\Delta)$ 是端点窗参数, 且对所有充分大的 $\Delta$ 都有 $v_1(\Delta)\le u$ 和 $v_2(\Delta)\ge v$. 则
            }
            {
                \setlength\belowdisplayskip{-10pt}
                \begin{equation*}
                    \lim_{\Delta\to\infty}
                    \frac{\Delta^2}{L_\Delta}
                    \log\mathbb P\left(
                        \begin{aligned}
                            -a\Delta\le Y_j\le (1-a)\Delta,\,
                            \forall j\in[\![1,L_\Delta]\!],
                            \\ (v_1-a)\Delta\le Y_{L_\Delta}\le(v_2-a)\Delta
                        \end{aligned}
                    \right)
                    =
                    -\frac{\pi^2}{2}\sigma^2.
                \end{equation*}
            }
    \end{enumerate}
\end{lemma}
\begin{proof}
    \bilingual{
        For (i), we prove here only the result for $x$. The result for $y$ is the same since the conditions for \nameref{theorem:mogulskii} are satisfied by both $x$ and $y$ as stated in \Cref{lemma:mogulskii-x-y}.
    }{
        对于 (i), 由于$x$ 和 $y$ 都满足 \nameref{theorem:mogulskii}的条件, 这里只对$x$证明, 对$y$的证法相同.
    }
    \begin{enumerate}[label=(\roman*)]
        \item
            \bilingual{
                We denote here the step random variables of $(Y_i)$ as $(\sigma\xi_i)$. Define $s_n$ as in \eqref{eq:mogulskii-s-n-t}. We can express $q(x(n),n-1)$ in $s_n$ as
            }{
                记 $(Y_i)$ 的步长随机变量为 $(\sigma\xi_i)$. 定义 $s_n$ 如 \eqref{eq:mogulskii-s-n-t}. 可将 $q(x(n),n-1)$ 表示为 $s_n$ 的函数:
            }
            \begin{align*}
                q(x(n),n-1)
                &=
                \log\mathbb P
                \left(
                    \frac 1\sigma(-a)\le s_n(t)\le\frac 1\sigma(1-a),\,
                    t\in[0,1]
                \right) \nonumber
                \\ &=:\log\mathbb P(s_n\in G_0),
                \quad n\ge 1.
            \end{align*}
            \bilingual{
                For each $m\ge 1$, we apply \nameref{theorem:mogulskii} for $G=G(m)$ of the form \eqref{eq:mogulskii-G} with
            }{
                对于每个 $m\ge 1$, 将 \nameref{theorem:mogulskii}应用于如下形式的 $G=G(m)$:
            }
            \begin{equation*}
                S_1=\frac{1}\sigma\left(1-a-\frac 1{2m}\right),\,
                S_2=\frac{1}\sigma\left(-a+\frac 1{2m}\right),
            \end{equation*}
            \bilingual{
                and we can use \eqref{eq:mogulskii-h-alpha} to calculate the corresponding $H_2(G)$ as
            }{
                并且可以使用 \eqref{eq:mogulskii-h-alpha} 来计算对应的 $H_2(G)$:
            }
            \begin{equation*}
                H_2(G)=\sigma^2\left(1-\frac 1m\right)^{-2}.
            \end{equation*}
            \bilingual{
                Hence, by \nameref{theorem:mogulskii}, we have
            }{
                因此, 根据 \nameref{theorem:mogulskii}, 有
            }
            \begin{align*}
                \lim_{n\to\infty}
                \frac{x^2(n)}{n}
                \log\mathbb P(s_n\in G(m))
                =
                -\frac{\pi^2}2\sigma^2\left(1-\frac 1m\right)^{-2}.
            \end{align*}
            \bilingual{
                Since $G(m)\subset G_0$, by monotonicity of probability,
            }{
                由于 $G(m)\subset G_0$, 根据概率的单调性,
            }
            \begin{align*}
                \liminf_{n\to\infty}p(x(n),n-1)
                &\ge\liminf_{n\to\infty}\frac{x^2(n)}n\log\mathbb P(s_n\in G(m))
                \nonumber\\ &
                =-\frac{\pi^2}2\sigma^2\left(1-\frac 1m\right)^{-2}.
            \end{align*}
            \bilingual{
                Let $m\to\infty$ and we get
            }{
                令 $m\to\infty$, 可得
            }
            \begin{equation*}
                \liminf_{n\to\infty}p(x(n),n-1)\ge-\frac{\pi^2}{2}\sigma^2.
            \end{equation*}
            \bilingual{
                Similarly, we apply \nameref{theorem:mogulskii} for $G=\widetilde G(m)$ of the form \eqref{eq:mogulskii-G} with
            }{
                同样地, 将 \nameref{theorem:mogulskii}应用于如下形式的 $G=\widetilde G(m)$:
            }
            \begin{equation*}
                S_1=\frac{1}\sigma\left(1-a+\frac 1{2m}\right),\,
                S_2=\frac{1}\sigma\left(-a-\frac 1{2m}\right),
            \end{equation*}
            \bilingual{
                and we can get
            }{
                从而可以得到
            }
            \begin{equation*}
                \limsup_{n\to\infty}p(x(n),n-1)\le-\frac{\pi^2}2\sigma^2\left(1+\frac 1m\right)^{-2}.
            \end{equation*}
            \bilingual{
                Let $m\to\infty$ and the result would follow.
            }{
                令 $m\to\infty$, 结论得证.
            }
        \item
            \bilingual{
                By Lemma \ref{lemma:mogulskii-mono} and since $x(n)\le\Delta\le x(n+1)-1=y(n)$, we have the lower and upper bound for $p(\Delta,L_\Delta)$ as
            }{
                根据引理 \ref{lemma:mogulskii-mono} 并且由于 $x(n)\le\Delta\le x(n+1)-1=y(n)$, 可得 $p(\Delta,L_\Delta)$ 的上下界为
            }
            \begin{align*}
                p(\Delta,L_\Delta)
                &=\frac{\Delta^2}{L_\Delta}q(\Delta,L_\Delta)
                \ge\frac{\Delta^2}{L_\Delta}q(x(n),L_\Delta)
                =\frac{\Delta^2}{x^2(n)}p(x(n),n-1)
                \\
                p(\Delta,L_\Delta)
                &=\frac{\Delta^2}{L_\Delta}q(\Delta,L_\Delta)
                \le\frac{\Delta^2}{L_\Delta}q(y(n),L_\Delta)
                =\frac{\Delta^2}{y^2(n)}p(y(n),n-1)
            \end{align*}
            \bilingual{
                Let $\Delta\to\infty$ in both equations and it suffices to show that
            }{
                在两式中令 $\Delta\to\infty$, 只需证明
            }
            \begin{equation*}
                \lim_{\Delta\to\infty}\frac{\Delta}{x(n)}
                =\lim_{\Delta\to\infty}\frac{\Delta}{y(n)}
                =1.
            \end{equation*}
            \bilingual{
                Indeed, we have $x(n)\le\Delta\le x(n)+N_\Delta$ and $y(n)-N_\Delta\le\Delta\le y(n)$, and the result follows from the additional condition.
            }{
                事实上, 由于 $x(n)\le\Delta\le x(n)+N_\Delta$ 和 $y(n)-N_\Delta\le\Delta\le y(n)$, 结论得证.
            }
        \item
            \bilingual{
                First fix $0<u<v<1$ and consider the endpoint window with these fixed fractions of the tube width. We prove its rate using one fixed set $G$ in \nameref{theorem:mogulskii}; the set will not depend on $n$.
            }{
                先固定 $0<u<v<1$, 考虑端点比例固定为这两个数的管道端点窗. 我们用 \nameref{theorem:mogulskii} 中的一个固定集合 $G$ 求得其概率率; 该集合不依赖于 $n$.
            }
            \bilingual{
                We prove the limit by contradiction. If it failed, there would be a sequence $\Delta_k\to\infty$ along which the normalized logarithm stays a fixed positive distance from the claimed limit. Since $L_k:=L_{\Delta_k}$ satisfies $L_k/\Delta_k^2\to\infty$, pass to a subsequence on which $L_k$ is strictly increasing, and put $n_k:=L_k+1$. Define an auxiliary scale sequence $\widehat x(n)$ by $\widehat x(n)=\Delta_k$ for $n_k\le n<n_{k+1}$, with arbitrary positive values before $n_1$. Then $\widehat x(n)\to\infty$ and $\widehat x(n)/\sqrt n\to0$, because $\Delta_k^2/n_k\to0$.
            }{
                用反证法证明该极限. 若极限不成立, 则存在趋于无穷的序列 $(\Delta_k)$, 使相应的归一化对数概率与所称极限始终相差一个固定正数. 由于 $L_k:=L_{\Delta_k}$ 满足 $L_k/\Delta_k^2\to\infty$, 取子列后可设 $L_k$ 严格递增, 并令 $n_k:=L_k+1$. 定义辅助尺度序列 $\widehat x(n)$: 当 $n_k\le n<n_{k+1}$ 时令 $\widehat x(n)=\Delta_k$, 在 $n_1$ 之前任取正值. 因为 $\Delta_k^2/n_k\to0$, 故 $\widehat x(n)\to\infty$ 且 $\widehat x(n)/\sqrt n\to0$.
            }
            \begin{equation*}
                S_2(t)=
                \begin{cases}
                    -a/\sigma, &t\in[0,1),\\
                    (u-a)/\sigma, &t=1,
                \end{cases}
                \qquad
                S_1(t)=
                \begin{cases}
                    (1-a)/\sigma, &t\in[0,1),\\
                    (v-a)/\sigma, &t=1.
                \end{cases}
            \end{equation*}
            \bilingual{
                Let $G$ be the path class \eqref{eq:mogulskii-G} for these fixed boundaries. They are right-continuous and piecewise constant, and satisfy \eqref{eq:mogulskii-S-1-S-2}: at $t=1$, the common admissible overlap is $((u-a)/\sigma,(v-a)/\sigma)$, which is nonempty. Its width is $1/\sigma$ for almost every $t$, so $H_2(G)=\sigma^2$.
            }{
                令 $G$ 为以上固定边界对应的 \eqref{eq:mogulskii-G} 路径类. 这些边界右连续且分段常数, 并满足 \eqref{eq:mogulskii-S-1-S-2}: 在 $t=1$ 处, 共同可行区间为 $((u-a)/\sigma,(v-a)/\sigma)$, 因而非空. 边界宽度在几乎处处等于 $1/\sigma$, 故 $H_2(G)=\sigma^2$.
            }
            \bilingual{
                By the terminal-left convention for $s_n$, $s_n(1)=\sigma^{-1}\sum_{j=1}^{n-1}\xi_j=Y_{n-1}/(\sigma\widehat x(n))$. Thus for $n=n_k=L_{\Delta_k}+1$, the source endpoint is exactly $Y_{L_{\Delta_k}}/(\sigma\Delta_k)$. Therefore
            }{
                按照 $s_n$ 在终点左连续的约定, $s_n(1)=\sigma^{-1}\sum_{j=1}^{n-1}\xi_j=Y_{n-1}/(\sigma\widehat x(n))$. 因而在 $n=n_k=L_{\Delta_k}+1$ 时, 源路径终点恰为 $Y_{L_{\Delta_k}}/(\sigma\Delta_k)$. 所以
            }
            \begin{align*}
                \mathbb P(s_{n_k}\in G)
                &\le
                \mathbb P\left(
                    \begin{aligned}
                        -a\Delta_k\le Y_j\le(1-a)\Delta_k,\,
                        &\forall j\in[\![1,L_k]\!],\\
                        &(u-a)\Delta_k\le Y_{L_k}\le(v-a)\Delta_k
                    \end{aligned}
                \right)\\
                &\le
                \mathbb P\left(
                    -a\Delta_k\le Y_j\le(1-a)\Delta_k,\,
                    \forall j\in[\![1,L_k]\!]
                \right).
            \end{align*}
            \bilingual{
                Mogul'ski\u i's theorem for this fixed $G$, together with
                $n_k/L_k\to1$, gives the limit
            }{
                对这个固定的 $G$ 应用 Mogul'ski\u i 定理, 并利用
                $n_k/L_k\to1$, 得到以下极限:
            }
            \begin{equation*}
                \frac{\Delta_k^2}{L_k}
                \log\mathbb P(s_{n_k}\in G)
                \longrightarrow -\frac{\pi^2}{2}\sigma^2.
            \end{equation*}
            \bilingual{
                For the upper bound, enlarge the horizontal tube by
                $\varepsilon>0$ on both sides. Its width becomes
                $1+2\varepsilon$, so its rate tends to
                $-\frac{\pi^2}{2}\sigma^2$ as $\varepsilon\downarrow0$.
                Squeezing gives the fixed-window limit along the chosen
                subsequence, a contradiction. The same argument applies to
                arbitrary sequences $\Delta_k\to\infty$ and
                $L_k/\Delta_k^2\to\infty$, provided the endpoint windows
                eventually contain a common interval $[u,v]$. Thus the
                convergence is uniform over the block choices below.
            }{
                为得到上界, 将水平管道两侧各放宽 $\varepsilon>0$. 新管道宽度为
                $1+2\varepsilon$, 故其速率在 $\varepsilon\downarrow0$ 时趋于
                $-\frac{\pi^2}{2}\sigma^2$. 由夹逼, 所选子列上固定端点窗的极限
                成立, 矛盾. 同一论证适用于任意满足 $\Delta_k\to\infty$ 与
                $L_k/\Delta_k^2\to\infty$ 的序列, 只要端点窗最终包含一个共同区间
                $[u,v]$. 因而该收敛对下文的分块选择是一致的.
            }
            \begin{equation*}
                \lim_{\Delta\to\infty}
                \frac{\Delta^2}{L_\Delta}
                \log\mathbb P\left(
                    \begin{aligned}
                        -a\Delta\le Y_j\le(1-a)\Delta,\,
                        \forall j\in[\![1,L_\Delta]\!],\\
                        (u-a)\Delta\le Y_{L_\Delta}\le(v-a)\Delta
                    \end{aligned}
                \right)
                =-\frac{\pi^2}{2}\sigma^2.
            \end{equation*}
            \bilingual{
                Finally, the eventual inclusions $v_1(\Delta)\le u<v\le v_2(\Delta)$ sandwich the variable-window event between this fixed-window event and the horizontal-tube event. The fixed-window lower rate was proved above, while the horizontal-tube upper rate follows from the enlarged fixed path classes used there. Their logarithmic rates agree, proving the stated conclusion.
            }{
                最后, 对充分大的 $\Delta$, 条件 $v_1(\Delta)\le u<v\le v_2(\Delta)$ 使变动端点窗事件夹在上述固定端点窗事件与水平管道事件之间. 固定端点窗的下界率已在上文证明, 而水平管道的上界率由上文放宽后的固定路径类给出. 两者的对数率相同, 故得所述结论.
            }
        \qedhere
    \end{enumerate}
\end{proof}
