\section{主要定理的证明}

\bilingual{
    This section is devoted to the proof of the asymptotic speed order of the minimum position of the $N$-branching random walk with infinite offspring. The assumptions in this section include \Cref{assumption:psi-zero,assumption:at-least-one-child,assumption:psi-one,assumption:psi-prime-one,assumption:finite-variance,assumption:log-expt-cross-sum-exp-neg-sum,assumption:moment-1,assumption:moment-4,assumption:moment-exp-1},~ i.e., $\psi(0)>0$ (supercritical regime), $\#\Xi\ge 1$ a.s. (at least one child), $\psi(1)=0$ (normalization), $\psi'(1)=0$ (zero mean of the spine process), $\mathbb E[\xi^2]=\sigma^2<\infty$ (finite variance of the spine process), $\log\mathbb E[\sum_{i,j\in\mathbb N, i\neq j} e^{-(\Xi_i+\Xi_j)}]<\infty$ (finite cross log-Laplacian), $\mathbb E[(\Xi_1)_+]<\infty$ (finite positive part of the leftmost step), $\mathbb E[(\Xi_1)_+^4]<\infty$ (finite fourth moment of the positive part of the leftmost step), and there exists some $\lambda>0$ such that $\mathbb E[e^{\lambda \Xi_1}]<\infty$ (finite exponential moment of the leftmost step).
}{
    本节将给出无限分支 $N$-分支随机游走最左侧粒子位置渐进速率阶数的证明. 本节的假设包括\Cref{assumption:psi-zero,assumption:at-least-one-child,assumption:psi-one,assumption:psi-prime-one,assumption:finite-variance,assumption:log-expt-cross-sum-exp-neg-sum,assumption:moment-1,assumption:moment-4,assumption:moment-exp-1},~ 即$\psi(0)>0$ (超临界情形), $\#\Xi\ge 1$ a.s. (至少一个子代), $\psi(1)=0$ (标准化), $\psi'(1)=0$ (脊柱过程零均值), $\mathbb E[\xi^2]=\sigma^2<\infty$ (脊柱过程有限方差), $\log\mathbb E[\sum_{i,j\in\mathbb N, i\neq j} e^{-(\Xi_i+\Xi_j)}]<\infty$ (交叉log-Laplacian有限), $\mathbb E[(\Xi_1)_+]<\infty$ (最左侧正部有限), $\mathbb E[(\Xi_1)_+^4]<\infty$ (最左侧正部四阶矩有限), 并且存在某个 $\lambda>0$ 使得 $\mathbb E[e^{\lambda \Xi_1}]<\infty$ (最左侧指数矩有限).
}

\bilingual{
    First, by using the several lemmas proved in the previous section, we show a stronger $\mathrm L^2$ convergence result as a tool for proving the main theorem. This generalizes Proposition 3.1 in A\"id\'ekon and Hu \mycite{aïdékon2025averageweightpercolationcompletegraph} to apply to more general assumptions.
}{
    首先,~ 利用上文证得的多个引理,~ 证明一个更强的$\mathrm L^2$收敛的工具性结论.~ 这推广了 A\"id\'ekon 与 Hu \mycite{aïdékon2025averageweightpercolationcompletegraph} 中的命题 3.1,~ 以适用于更一般的假设条件.
}

\begin{theorem}
    \label{proposition:3.1}
    \bilingual{
        Fix $a\in[0,1]$ and $t>0$. If $\mathcal V^N=\mathcal V^{N|a}$ starts with $\lfloor N^a\rfloor$ particles located at $0$, then for any sequence $\ell_N$ satisfying $\lim_{N\to\infty}\frac{\ell_N}{\log^3N}=t$, 
    }{
        固定 $a\in[0,1]$ 和 $t>0$. 若 $\mathcal V^N=\mathcal V^{N|a}$ 从 $\lfloor N^a\rfloor$ 个位于 $0$ 的粒子开始,~ 那么对于任何满足 $\lim_{N\to\infty}\frac{\ell_N}{\log^3N}=t$ 的数列 $\ell_N$,
    }
    {
        \setlength{\belowdisplayskip}{0pt}
        \begin{equation}
            \frac 1{\log N}
            \min_{|u|=\ell_N}
            \mathcal V^N(u)
            \xrightarrow{\mathrm L^2}
            -a+\frac{\pi^2}2\sigma^2t.
        \end{equation}
    }
\end{theorem}

\bilingual{
    The meaning of this theorem is that, if we take the time axis to be of order $\log^3 N$ (i.e., time grows with $N$ at a triple logarithmic rate) and the one-dimensional space axis to be of order $\log N$, then, in this non-symmetrically logarithmically scaled time-space, there exists a line $-a+\frac{\pi^2}2\sigma^2t$, such that the minimum position of the $N$-branching random walk, at each vertical time slice $t$, converges in $\mathrm L^2$ sense to the point on the aforementioned line at that slice as $N$ increases.
}{
    这个定理的含义是说,~ 把时间轴取为 $\log^3 N$ 的数量级 (即时间随着 $N$ 增加而以三次对数的速率增长),~ 把一维空间轴取为 $\log N$ 的数量级,~ 那么,~ 在这个经不对称对数缩放的时空中,~ 存在一条直线$-a+\frac{\pi^2}2\sigma^2t$,~ 使得$N$-分支随机游走的最左侧位置,~ 在每一个纵向的时间切片 $t$ 处,~ 随着$N$增加而在$\mathrm L^2$ 的意义下收敛到上述直线的该时间切片处的点.
}

\begin{proof}[\bilingual{Proof of \Cref{proposition:3.1}}{\Cref{proposition:3.1} 的证明}]
    \quad

    \bilingual{
        The result would follow if we can show that
    }{
        命题将得证,~ 若能证得
    }
    \begin{align}
        \lim_{N\to\infty}
        \frac 1{\log N}
        \mathbb E\left[
            \min_{|u|=\ell_N}\mathcal V^N(u)
        \right]
        &
        =
        -a+\frac{\pi^2}2\sigma^2t,
        \label{eq:4.13}
        \\
        \lim_{N\to\infty}
        \frac 1{\log^2 N}
        \mathrm{Var}\!\left[
            \min_{|u|=\ell_N}\mathcal V^N(u)
        \right]
        &
        =
        0.
        \label{eq:4.14}
    \end{align}
    \bilingual{
        Indeed, for
    }{
        事实上,~ 对
    }
    \begin{equation*}
        \mathcal M^N_n:=\min_{|u|=n}\mathcal V^N(u), \quad
        \mathcal X_N:=\frac 1{\log N}\mathcal M^N_{\ell_N}, \quad
        \mu=\mu(t):=-a+\frac{\pi^2}2\sigma^2t,
    \end{equation*}
    \bilingual{
        we have
    }{
        有
    }
    \begin{equation*}
        \mathbb E\left[
            \left(
                \mathcal X_N-\mu
            \right)^2
        \right]
        =
        \mathrm{Var}\!\left[
            \mathcal X_N
        \right]
        +
        \left(
            \mathbb E\left[
                \mathcal X_N
            \right]
            -\mu
        \right)^2
        \xrightarrow{}
        0,
        \quad
        N\to\infty.
    \end{equation*}

    \quad

    \bilingual{
        Further, it suffices to show that for any $\zeta<\mu<\zeta'$, there exists some $\varepsilon=\varepsilon(\zeta,\zeta')>0$ such that for some $N_0=N_0(\varepsilon,\zeta,\zeta')>0$ and all $N\ge N_0$,
    }{
        进一步,~ 只需证明对于任意 $\zeta<\mu<\zeta'$,~ 存在某个 $\varepsilon=\varepsilon(\zeta,\zeta')>0$ 使得对于某个 $N_0=N_0(\varepsilon,\zeta,\zeta')>0$ 和所有 $N\ge N_0$,
    }
    \begin{align}
        \mathbb P(\mathcal X_N\ge\zeta')
        =
        \mathbb P\left(
            \mathcal M^N_{\ell_N}
            \ge
            \zeta'\log N
        \right)
        &
        \le
        N^{-\varepsilon},
        \label{eq:4.15}
        \\
        \mathbb P(\mathcal X_N\le\zeta)
        =
        \mathbb P\left(
            \mathcal M^N_{\ell_N}
            \le
            \zeta\log N
        \right)
        &
        \le
        N^{-\varepsilon}.
        \label{eq:4.16}
    \end{align}
    \bilingual{
        Indeed, observe that $\mathcal M^N_n$ is (stochastically) smaller than $\mathcal V^1$ at time $n$, where $\mathcal V^1$ is the branching random walk in which only the leftmost particle is kept at each step. Therefore, $\mathcal V^1$ is a random walk with step distribution $\Xi_1$. Under \Cref{assumption:psi-one,assumption:moment-4} (i.e. $\psi(1)=0$ and $\mathbb E[(\Xi_1)_+^4]<\infty$), we can show that
    }{
        事实上,~ 注意到 $\mathcal M^N_n$ 小于 (或随机受制于) $\mathcal V^1$ 在时刻 $n$ 的位置,~ 其中 $\mathcal V^1$ 是每一步只保留最左侧粒子的分支随机游走.~ 因此,~ $\mathcal V^1$ 是一个步长分布为 $\Xi_1$ 的随机游走.~ 在\Cref{assumption:psi-one,assumption:moment-4} (即 $\psi(1)=0$ 和 $\mathbb E[(\Xi_1)_+^4]<\infty$) 下,~ 我们可以证明
    }
    \begin{equation*}
        \mathbb E[(\mathcal M^N_{n})_-^4]=O_n(1),
        \quad
        \mathbb E[(\mathcal M^N_{n})_+^4]=O_n(n^4),
    \end{equation*}
    \bilingual{
        as $n$ goes to infinity (see \Cref{lemma:M:moment-4:neg,lemma:M:moment-4:pos} for details), and thus
    }{
        在 $n$ 趋近于无穷大时成立 (详见\Cref{lemma:M:moment-4:neg,lemma:M:moment-4:pos}),~ 从而
    }
    \begin{equation*}
        \mathbb E[(\mathcal M^N_{n})^4]
        \le
        \mathbb E[(\mathcal M^N_{n})_-^4]
        +
        \mathbb E[(\mathcal M^N_{n})_+^4]
        =
        O_n(n^4).
    \end{equation*}
    \bilingual{
        Now, let for $N\ge 1$,
    }{
        现在,~ 对 $N\ge 1$,~ 令
    }
    \begin{equation*}
        H_N
        :=\left\{
            \zeta\log N\le \mathcal M^N_{\ell_N}\le \zeta'\log N
        \right\}
        =
        \left\{
            \zeta\le \mathcal X_N\le \zeta'
        \right\}.
    \end{equation*}
    \bilingual{
        According to \eqref{eq:4.15} and \eqref{eq:4.16}, for all $N\ge N_0$,
    }{
        根据\eqref{eq:4.15} 和 \eqref{eq:4.16},~ 对所有 $N\ge N_0$,
    }
    $
        \mathbb P(
            H_N^{\mathrm c}
        )
        \le
        2N^{-\varepsilon}
    $,
    \bilingual{
        and thus
    }{
        从而
    }
    \begin{equation*}
        \mathbb P(
            H_N^{\mathrm c}
        )
        =
        O_N(N^{-\varepsilon}),
        \quad
        \lim_{N\to\infty}\mathbb P(H_N)=1.
    \end{equation*}
    \bilingual{
        By Cauchy--Schwarz inequality and the fact that $\ell_N=O_N(\log^3 N)$, we have
    }{
        根据 Cauchy--Schwarz 不等式和 $\ell_N=O_N(\log^3 N)$ 的事实,~ 有
    }
    \begin{align*}
        \mathbb E\left[
            \mathcal X_N^2
            {\bf 1}_{H_N^{\mathrm c}}
        \right]
        &
        \le
        \frac{
            \mathbb E\left[
                \left(
                    \mathcal M^N_{\ell_N}
                \right)^4
            \right]^{\frac 12}
            \!\!\!
            \cdot
            \mathbb P\!\left(
                H_N^{\mathrm c}
            \right)^{\frac 12}
        }{\log^2 N}
        =
        \frac{
            O_N(\ell_N^2)
            \cdot
            O_N(N^{-\varepsilon/2})
        }{\log^2 N}
        =
        o_N(1),
    \end{align*}
    \bilingual{
        and thus
    }{
        从而
    }
    \begin{equation*}
        \left|
            \mathbb E\left[
                \mathcal X_N
                {\bf 1}_{H_N^{\mathrm c}}
            \right]
        \right|
        \le
        \mathbb E\left[
            |\mathcal X_N|
            {\bf 1}_{H_N^{\mathrm c}}
        \right]
        \le
        \mathbb E\left[
            \mathcal X_N^2
            {\bf 1}_{H_N^{\mathrm c}}
        \right]^{\frac 12}
        =
        o_N(1).
    \end{equation*}
    \bilingual{
        On the other hand, by definition of $H_N$,
    }{
        另一方面,~ 根据 $H_N$ 的定义,
    }
    \begin{align*}
        \zeta\mathbb P(H_N)
        \le
        \mathbb E\left[
            \mathcal X_N
            {\bf 1}_{H_N}
        \right]
        \le
        \zeta'\mathbb P(H_N),
        \\
        \zeta^2\mathbb P(H_N)
        \le
        \mathbb E\left[
            \mathcal X_N^2
            {\bf 1}_{H_N}
        \right]
        \le
        \zeta'^2\mathbb P(H_N),
    \end{align*}
    \bilingual{
        Let $N\to\infty$, then $\zeta,\zeta'\to \mu$, we get
    }{
        令 $N\to\infty$,~ 其次令 $\zeta,\zeta'\to \mu$,~ 得到
    }
    \begin{align*}
        \lim_{N\to\infty}
        \mathbb E[ \mathcal X_N ]
        =
        \lim_{N\to\infty}
        \mathbb E[ \mathcal X_N {\bf 1}_{H_N} ]
        &
        =
        \mu,\\
        \lim_{N\to\infty}
        \mathbb E[ \mathcal X_N^2 ]
        =
        \lim_{N\to\infty}
        \mathbb E[ \mathcal X_N^2 {\bf 1}_{H_N} ]
        &
        =
        \mu^2,
    \end{align*}
    \bilingual{
        yielding \eqref{eq:4.13} and \eqref{eq:4.14}.
    }{
        从而得到 \eqref{eq:4.13} 和 \eqref{eq:4.14}.
    }


    \quad

    \bilingual{
        We will first show \eqref{eq:4.15} for the case of $a\in(0,1)$ in the first sub-section, then for the case of $a=1$ and $a=0$ in the second sub-section, and finally show \eqref{eq:4.16} for all $a\in[0,1]$ in the third sub-section.
    }{
        本证明的剩余部分,~ 将首先在第一小节中证明 \eqref{eq:4.15} 在 $a\in(0,1)$ 的情形下成立,~ 然后在第二小节中证明对于 $a=1$ 和 $a=0$ 的情形成立,~ 最后在第三小节中证明 \eqref{eq:4.16} 对于所有 $a\in[0,1]$ 的情形成立.
    }
\end{proof}

\subsection{\bilingual{Proof of \texorpdfstring{\eqref{eq:4.15}}{First Equation} for \texorpdfstring{$a\in(0,1)$}{a∈(0,1)}}{在 \texorpdfstring{$a\in(0,1)$}{a∈(0,1)} 的情形下右侧多项式衰减的证明}}

\begin{proof}[\bilingual{Proof of \eqref{eq:4.15} for the case of \texorpdfstring{$a\in(0,1)$}{a∈(0,1)}}{在 $a\in(0,1)$ 的情形下 \eqref{eq:4.15} 的证明}]
    \quad

    \bilingual{
        We now proceed to show \eqref{eq:4.15} for the case of $a\in(0,1)$. 
    }{
        现在继续证明 \eqref{eq:4.15} 在 $a\in(0,1)$ 的情形下成立.
    }
    
    \bilingual{
        In light of \Cref{lemma:4.8}, we construct a killed branching random walk $\mathcal V^{\mathrm{k}}$ (killed according to a given rule specified later). Denote $\mathcal M^{\mathrm{k}}_n:=\min_{|u|=n}\mathcal V^{\mathrm{k}}(u)$ for $n\ge 0$. So long as $\mathcal V^{\mathrm{k}}$ satisfies the conditions listed in \Cref{lemma:4.8}, we have
    }{
        根据\Cref{lemma:4.8},~ 构造一个经灭杀的的分支随机游走 $\mathcal V^{\mathrm{k}}$ (灭杀规则见后文具体说明).~ 对 $n\ge 0$ 记 $\mathcal M^{\mathrm{k}}_n:=\min_{|u|=n}\mathcal V^{\mathrm{k}}(u)$.~ 只要 $\mathcal V^{\mathrm{k}}$ 满足\Cref{lemma:4.8} 中列出的条件,~ 就有
    }
    \begin{equation*}
        \mathcal M^N_{\ell_N}
        \le
        \mathcal M^{\mathrm{k}}_{\ell_N}
        \quad
        \text{
            \bilingual{
                on the event $\{\tau>\ell_N\}$,
            }{
                在事件 $\{\tau>\ell_N\}$ 上,
            }
        }
    \end{equation*}
    \bilingual{
        where $\tau$ is the first time that the generation size of $\mathcal V^{\mathrm{k}}$ exceeds $N$. Note that the left-hand side in \eqref{eq:4.15} 
    }{
        其中 $\tau$ 是 $\mathcal V^{\mathrm{k}}$ 的粒子总数首次超过 $N$ 的时刻. 注意到 \eqref{eq:4.15} 的左侧
    }
    \begin{align}
        &\quad
        \mathbb P\left(
            \mathcal M^N_{\ell_N}
            \ge
            \zeta'\log N
        \right)
        \nonumber
        \\
        &
        =
        \mathbb P\left(
            \mathcal M^N_{\ell_N}
            \ge
            \zeta'\log N,\,
            \tau\le\ell_N
        \right)
        +
        \mathbb P\left(
            \mathcal M^N_{\ell_N}
            \ge
            \zeta'\log N,\,
            \tau>\ell_N
        \right)
        \nonumber
        \\ &
        \le
        \mathbb P\left(
            \tau\le\ell_N
        \right)
        +
        \mathbb P\left(
            \mathcal M^{\mathrm{k}}_{\ell_N}
            \ge
            \zeta'\log N
        \right).
        \label{eq:prop:3.1:1}
    \end{align}
    \bilingual{
        Therefore, we need to construct a $\mathcal V^{\mathrm{k}}$ such that, with high probability,
    }{
        因此,~ 需要构造一个 $\mathcal V^{\mathrm{k}}$,~ 使得,~ 以高概率,
    }
    \begin{itemize}
        \item 
        \bilingual{
            $\tau>\ell_N$, i.e., the generation size of $\mathcal V^{\mathrm{k}}$ does not exeed $N$ till time $\ell_N$;
        }{
            $\tau>\ell_N$,~ 即 $\mathcal V^{\mathrm{k}}$ 的粒子总数在时刻 $\ell_N$ 之前不超过 $N$;
        }
        \item 
        \bilingual{
            $\mathcal M^{\mathrm{k}}_{\ell_N}<\zeta'\log N$, i.e., the minimum position of $\mathcal V^{\mathrm{k}}$ is below $\zeta'\log N$ at time $\ell_N$;
        }{
            $\mathcal M^{\mathrm{k}}_{\ell_N}<\zeta'\log N$,~ 即 $\mathcal V^{\mathrm{k}}$ 的最左侧位置在时刻 $\ell_N$ 时低于 $\zeta'\log N$;
        }
    \end{itemize}
    \bilingual{
        and, the probability of deviating from these two normal situations, i.e., the right-hand side of \eqref{eq:prop:3.1:1}, decays at least polynomially with respect to $N$. Note that, the left-hand side of \eqref{eq:prop:3.1:1} is naturally hard to compute, since the selection mechanism underlying the $N$-BRW requires that we keep track of the whole population at each step, meaning that when we are determining whether a particular particle should be preserved, we have to look at the positions of all other particles, including the descendants of different ancesters at time $0$, which is hard to analyze. On the other hand, the right-hand side of \eqref{eq:prop:3.1:1} is easier to compute, since we can construct $\mathcal V^{\mathrm{k}}$ with a simpler selection mechanism, so that when we are determining whether a particular particle should be preserved, we only need to look at the position of itself, and thus we can analyze $\mathcal V^{\mathrm{k}}$ by using the many-to-one formula and Mogulskii's estimate for random walks.
    }{
        并且,~ 偏离上述两种正常情况的概率,~ 即 \eqref{eq:prop:3.1:1} 的右侧,~ 至少以 $N$ 的多项式速率衰减. 注意到,~ \eqref{eq:prop:3.1:1} 的左侧自然难以计算,~ 因为 $N$-分支随机游走底层的选择机制要求我们在每一步都要跟踪整个种群,~ 意味着当我们在确定一个特定粒子是否应该被保留时,~ 我们必须查看所有其他粒子的位置,~ 包括时刻 $0$ 不同祖先的后代的位置,~ 这很难分析. 另一方面,~ \eqref{eq:prop:3.1:1} 的右侧更容易计算,~ 因为我们可以构造一个具有更简单选择机制的 $\mathcal V^{\mathrm{k}}$,~ 使得当我们在确定一个特定粒子是否应该被保留时,~ 我们只需要查看它自己的位置,~ 从而我们可以使用多对一变换公式和 Mogulskii 对随机游走的估计来分析 $\mathcal V^{\mathrm{k}}$.
    }

    \bilingual{
        We first state our construction of the killed branching random walk $\mathcal V^{\mathrm{k}}$. Choose and fix $c\in(0,t^{-1/3})$, $b\in(0,a)$, and $v>0$ such that
    }{
        首先给出经灭杀的分支随机游走 $\mathcal V^{\mathrm{k}}$ 的构造.~ 选择并固定 $c\in(0,t^{-1/3})$, $b\in(0,a)$, 和 $v>0$ 使得
    }
    \begin{equation*}
        ct^{\frac 13}+(a-b)+v<1,\quad
        -b+\frac{\pi^2}2\sigma^2\left(c^{-2}t^{\frac 13}\right)+2v\left(1+t^{\frac 13}\right)<\zeta'.
    \end{equation*}
    \bilingual{
        Note that these inequalites are possible as $c\uparrow t^{-1/3}$, $b\uparrow a$, and $v\downarrow 0$. Also note that they are dependant on $a$, $t$, and $\zeta'$.
    }{
        注意到,~ 随着 $c\uparrow t^{-1/3}$,~ $b\uparrow a$,~ 和 $v\downarrow 0$,~ 这些不等式是可能满足的. 还要注意到它们依赖于 $a$,~ $t$,~ 和 $\zeta'$.
    }
    \begin{itemize}
        \item 
        \bilingual{
            At time $0$, start with $\lfloor N^a\rfloor$ particles located at $0$; and they branch accoding to $\Xi$; note that these make $\mathcal V^{\mathrm{k}}$ satisfy the condition of \Cref{lemma:4.8};
        }{
            在时刻 $0$,~ 从 $\lfloor N^a\rfloor$ 个位于 $0$ 的粒子开始;~ 它们根据分支律 $\Xi$ 进行分支;~ 注意到这些设定使得 $\mathcal V^{\mathrm{k}}$ 满足\Cref{lemma:4.8} 的条件;
        }
        \item 
        \bilingual{
            From time $0$ to $\ell:=\lfloor\log^{5/2}N\rfloor$, kill particles outside the tube $[-b\log N,0]$;
        }{
            从时刻 $0$ 到 $\ell:=\lfloor\log^{5/2}N\rfloor$,~ 灭杀管道 $[-b\log N,0]$ 以外的粒子;
        }
        \item
        \bilingual{
            At time $\ell$, further kill particles outside the interval $[-b\log N, (-b+v)\log N]$;
        }{
            在时刻 $\ell$,~ 进一步灭杀区间 $[-b\log N, (-b+v)\log N]$ 以外的粒子;
        }
        \item
        \bilingual{
            From time $\ell+1$ to $\ell_N$, for any alive particle in the last step at position $y$, kill particles outside a tilted tube with a small drift, i.e., $y+[r_j,s_j]$ for $j\in[\![\ell,\ell_N]\!]$, where $r_j:=\frac{\pi^2}{2}\sigma^2c^{-2}L^{-2/3}(j-\ell)$, $s_j:=r_j+cL^{1/3}$ and $L:=\ell_N-\ell$.
        }{
            从时刻 $\ell+1$ 到 $\ell_N$,~ 对于上一步中位置为 $y$ 的任何存活粒子,~ 灭杀一个带有小漂移的倾斜管道以外的粒子,~ 即 $y+[r_j,s_j]$ 对 $j\in[\![\ell,\ell_N]\!]$,~ 其中 $r_j:=\frac{\pi^2}{2}\sigma^2c^{-2}L^{-2/3}(j-\ell)$,~ $s_j:=r_j+cL^{1/3}$ 和 $L:=\ell_N-\ell$.
        }
        \item

        其中,~ 通过应用\Cref{lemma:2.3} 可得,~ 最后一个不等式对于某个 $N_1=N_1(a,t,\zeta)>0$ 和所有 $N\ge N_1$ 成立,~ 其中参数的选择为 $\Delta=b\log N$,~ $L=\ell_N$,~ $0<c=\frac b{b+\zeta+a}<1$ 以及任意 $d\in(0,\frac{\pi^2}2\sigma^2)$. ~另外可能需要确保当 $N\ge N_1$ 时 $L\ge(b\log N)^{2+c}$. ~事实上,~ 可以选择 $b$ 足够接近 $1$ 且 $d$ 足够接近 $\frac{\pi^2}2\sigma^2$ 以使得 $a+\zeta-dtb^{-2}<0$. ~因此,
    }
    \begin{equation*}
        \mathbb P\left(
            \mathcal M^{N|a}_{\ell_N}
            \ge
            \zeta\log N
        \right)
        \le
        N^{-(b-1)+o_N(1)}
        +
        N^{a+\zeta-dt b^{-2}+o_N(1)}
        \le
        N^{-\varepsilon},
    \end{equation*}
    \bilingual{
        where we chooce some $\varepsilon\in(0,\frac 12\min\{b-1,dtb^{-2}-a-\zeta\})$ such that $\varepsilon<b-1$ and $\varepsilon<dtb^{-2}-a-\zeta$, and it holds for some $N_0=N_0(a,t,\zeta)>N_1$ and all $N\ge N_0$. We are now done with the proof of \eqref{eq:4.16} for all $a\in[0,1]$.
    }{
        其中,~ 选择某个 $\varepsilon\in(0,\frac 12\min\{b-1,dtb^{-2}-a-\zeta\})$ 以使得 $\varepsilon<b-1$ 且 $\varepsilon<dtb^{-2}-a-\zeta$,~ 则对于某个 $N_0=N_0(a,t,\zeta)>N_1$ 和所有 $N\ge N_0$ 成立. ~至此,~ 对所有 $a\in[0,1]$ 的情形,~ \eqref{eq:4.16} 得证.
    }
\end{proof}

\subsection{\bilingual{Corollary on the asymptotic speed of the left-most particle}{关于最左侧粒子渐近速率的推论}}

\begin{theorem}
    \label{theorem:speed}
    \bilingual{
        Consider a branching random walk $\mathcal V^{N|a}$ that starts with $\lfloor N^a\rfloor$ particle at position $0$. Under all assumptions, the speed of the left-most particle in $\mathcal V^{N|a}$ exists and is given by
    }{
        考虑一个分支随机游走 $\mathcal V^{N|a}$,~ 它以 $\lfloor N^a\rfloor$ 个粒子在位置 $0$ 处开始. ~在所有假设下,~ $\mathcal V^{N|a}$ 中最左侧粒子的速率存在且由下式给出:
    }
    \begin{equation*}
        v_N
        =
        \lim_{n\to\infty}
        \frac{1}{n}\mathcal M^{N|a}_n
        =
        \frac{\pi^2}2\sigma^2\frac{1}{\log^2 N}
        +
        o_N\left(\frac{1}{\log^2 N}\right).
    \end{equation*}
\end{theorem}

\begin{proof}[\bilingual{Proof of \Cref{theorem:speed}}{\Cref{theorem:speed} 的证明}]
    \quad

    \bilingual{
        The existence of the speed is guaranteed by \Cref{corollary:leftmost-speed:N0} under all assumptions. We proceed to calculate the speed by showing that there exists an upper bound and a lower bound of the same order. For both bounds, we are to construct, with respect to $\mathcal V^{N|a}$, a branching random walk, $\mathcal V^{\mathrm{upper}}$ and $\mathcal V^{\mathrm{lower}}$ respectively, such that the minimum position in $\mathcal V^{\mathrm{upper}}$ is point-wise (or stochastically) larger than the minimum position in question, and the minimum position in $\mathcal V^{\mathrm{lower}}$ is point-wise (or stochastically) smaller than the minimum position in question. We then calculate the speed of the left-most particle in $\mathcal V^{\mathrm{upper}}$ and $\mathcal V^{\mathrm{lower}}$ using the proved result, \Cref{proposition:3.1}, to get the desired upper and lower bounds respectively.
    }{
        速率的存在性由\Cref{corollary:leftmost-speed:N0} 保证. ~现在通过构造上界和下界来计算速率. ~对于这两个界,~ 分别构造与 $\mathcal V^{N|a}$ 耦合的分支随机游走 $\mathcal V^{\mathrm{upper}}$ 和 $\mathcal V^{\mathrm{lower}}$,~ 使得 $\mathcal V^{\mathrm{upper}}$ 中的最小位置在点态大于（或随机占优于）所讨论的最小位置,~ 而 $\mathcal V^{\mathrm{lower}}$ 中的最小位置在点态小于（或随机受制于）所讨论的最小位置. ~然后使用已证明的结果\Cref{proposition:3.1} 来计算 $\mathcal V^{\mathrm{upper}}$ 和 $\mathcal V^{\mathrm{lower}}$ 中最左侧粒子的速率,~ 从而得到所需的上界和下界.
    }

    \bilingual{
        For any $t>0$, choose and fix some sequence $\ell_N=\ell_N(t)$ such that
    }{
        对于任意 $t>0$,~ 选择并固定某个序列 $\ell_N=\ell_N(t)$ 使得
    }
    \begin{equation*}
        \lim_{N\to\infty}\frac{\ell_N}{\log^3N}=t.
    \end{equation*}

    \bilingual{
        For the upper bound, we construct, with respect to $\mathcal V^{N|a}$, an upper branching random walk $\mathcal V^{\mathrm{upper}}$ as follows:
    }{
        对于上界,~ 构造与 $\mathcal V^{N|a}$ 耦合的上界分支随机游走 $\mathcal V^{\mathrm{upper}}$ 如下:
    }
    \begin{itemize}
        \item \bilingual{
            At time $0$, we kill all particles but one in the original $N$-BRW $\mathcal V^{N|a}$, so that there is only one particle at time $0$ and position $0$ in $\mathcal V^{\mathrm{upper}}$;
        }{
            在时刻 $0$,~ 除了一个粒子外,~ 在最初的 $N $-分支随机游走 $\mathcal V^{N|a}$ 中灭杀所有粒子,~ 使得 $\mathcal V^{\mathrm{upper}}$ 在零时刻只有一个粒子且位置为零;
        }
        \item \bilingual{
            From time $0$ till $\ell_N$, we let the process evolve as an $N$-BRW, i.e., at each step, we keep the left-most $N$ particles and kill all other particles;
        }{
            从时刻 $0$ 到 $\ell_N$,~ 让过程以 $N$-分支随机游走的方式演化,~ 即在每一步,~ 保留最左侧的 $N$ 个粒子并灭杀所有其他粒子;
        }
        \item \bilingual{
            At time $\ell_N$, we kill all particles but one left-most particle in $\mathcal V^{\mathrm{upper}}$;
        }{
            在时刻 $\ell_N$,~ 除了最左侧的一个粒子外,~ 在 $\mathcal V^{\mathrm{upper}}$ 中灭杀所有粒子;
        }
        \item \bilingual{
            From time $\ell_N$ onward, we let the process evolve as an $N$-BRW and repeat the killing procedure at all times $k\ell_N$ for $k\in\mathbb N$.
        }{
            从时刻 $\ell_N$ 开始,~ 让过程以 $N$-分支随机游走的方式演化,~ 并在所有时刻 $k\ell_N$ 处重复上述灭杀程序,~ 其中 $k\in\mathbb N$.
        }
    \end{itemize}
    \bilingual{
        By definition of $\mathcal V^{\mathrm{upper}}$, we have, by \Cref{corollary:2} and by induction on $k$, on the same probability space,
    }{
        根据 $\mathcal V^{\mathrm{upper}}$ 的定义有,~ 由\Cref{corollary:2} 和对 $k$ 的归纳法,~ 在相同的概率空间上,
    }
    \begin{equation*}
        \mathcal V^{\mathrm{upper}}_{k\ell_N}
        \succcurlyeq
        \mathcal V^{N|a}_{k\ell_N},
        \quad\forall k\in\mathbb N,
    \end{equation*}
    \bilingual{
        and, by some coupling of $\mathcal V^{\mathrm{upper}}$ and $\mathcal V^{N|a}$,
    }{
        并且,~ 通过 $\mathcal V^{\mathrm{upper}}$ 和 $\mathcal V^{N|a}$ 的某个耦合,
    }
    \begin{equation*}
        \mathcal V^{\mathrm{upper}}_n
        \succcurlyeq
        \mathcal V^{N|a}_n,
        \quad\forall n\in[\![k\ell_N,(k+1)\ell_N]\!],\,\forall k\in\mathbb N.
    \end{equation*}
    \bilingual{
        Hence, there is some coupling of $\mathcal V^{\mathrm{upper}}$ and $\mathcal V^{N|a}$ such that
    }{
        因此,~ 存在 $\mathcal V^{\mathrm{upper}}$ 和 $\mathcal V^{N|a}$ 的某个耦合使得
    }
    \begin{equation*}
        \mathcal M^{\mathrm{upper}}_n
        \ge
        \mathcal M^{N|a}_n,
        \quad\forall n\in\mathbb N.
    \end{equation*}
    \bilingual{
        Furthermore, by definition of $\mathcal V^{\mathrm{upper}}$, we have
    }{
        此外,~ 根据 $\mathcal V^{\mathrm{upper}}$ 的定义有
    }
    \begin{equation*}
        \mathcal V^{\mathrm{upper}}_{(k+1)\ell_N+n}
        -
        \mathcal M^{\mathrm{upper}}_{k\ell_N}
        \stackrel{\mathrm{d}}{=}
        \mathcal V^{N|0}_{\ell_N},
        \quad k\in\mathbb N,
    \end{equation*}
    \bilingual{
        and they are independent for different $k$'s, where, both sides of the equality is regarded as random finite counting measures, and the right-hand side is the population measure obtained from an $N$-BRW $\mathcal V^{N|0}$ starting with $N^0=1$ particle at position $0$ on the same probability space. Hence,
    }{
        并且对于不同的 $k$,~ 它们是独立的,~ 其中,~ 等式两边都被视为随机有限计数测度,~ 右侧是从一个 $N$-分支随机游走 $\mathcal V^{N|0}$ 中得到的粒子分布,~ 该 $N$-分支随机游走在相同的概率空间上以 $N^0=1$ 个粒子在位置 $0$ 处开始. ~因此,
    }
    \begin{equation*}
        \left(
            \min\left(
                \mathcal V^{\mathrm{upper}}_{(k+1)\ell_N}
                -
                \mathcal M^{\mathrm{upper}}_{k\ell_N}
            \right):
            k\in\mathbb N
        \right)
    \end{equation*}
    \bilingual{
        is an i.i.d.~sequence of random variables with the same distribution as $\min\mathcal V^{N|0}_{\ell_N}$. To use the Strong Law of Large Numbers (SLLN), we need to check that, given $N$,
    }{
        是一个独立同分布的随机变量序列,~ 其分布与 $\min\mathcal V^{N|0}_{\ell_N}$ 的分布相同. ~为了使用大数定律,~ 需要检查对于给定的 $N$,
    }
    \begin{equation*}
        \mathbb E\left[
            \min\mathcal V^{N|0}_{\ell_N}
        \right]
        =
        \mathbb E\left[
            \mathcal M^{N|0}_{\ell_N}
        \right]
        <
        \infty.
    \end{equation*}
    \bilingual{
        Indeed, under \Cref{assumption:psi-one,assumption:moment-1}, i.e., $\psi(1)=0$ and $\mathbb E[(\Xi_1)_+]<\infty$, we have
    }{
        事实上,~ 在\Cref{assumption:psi-one,assumption:moment-1} 下,~ 即 $\psi(1)=0$ 和 $\mathbb E[(\Xi_1)_+]<\infty$,~ 有
    }
    \begin{equation*}
        \left|
            \mathbb E\left[
                    \mathcal M^{N|0}_n
            \right]
        \right|
        \le
        \mathbb E\left[
            \left(
                \mathcal M^{N|0}_n
            \right)_+
        \right]
        +
        \mathbb E\left[
            \left(
                \mathcal M^{N|0}_n
            \right)_-
        \right]
        <
        \infty,
    \end{equation*}
    \bilingual{
        see \Cref{lemma:M:moment-1:neg,lemma:M:moment-1:pos} under the assumptions for details. Hence, by SLLN and the proven result of \Cref{proposition:3.1} for $a=0$,
    }{
        见\Cref{lemma:M:moment-1:neg,lemma:M:moment-1:pos} 在假设下的详细说明. ~因此,~ 通过大数定律和\Cref{proposition:3.1} 已证明的 $a=0$ 的结果,
    }
    \begin{align*}
        v_N
        &=
        \lim_{n\to\infty}
        \frac{1}{n}\mathcal M^{N|a}_n
        =
        \lim_{k\to\infty}
        \frac 1{k\ell_N}
        \mathcal M^{N|a}_{k\ell_N}
        \\ &
        \le
        \limsup_{k\to\infty}
        \frac 1{k\ell_N}
        \mathcal M^{\mathrm{upper}}_{k\ell_N}
        % \\ &
        =
        \frac 1{\ell_N}
        \limsup_{k\to\infty}
        \frac 1{k}
        \left(
            \sum_{i=1}^k
            \left(
                \mathcal M^{\mathrm{upper}}_{i\ell_N}
                -
                \mathcal M^{\mathrm{upper}}_{(i-1)\ell_N}
            \right)
        \right)
        \\ &
        =
        \frac 1{\ell_N}
        \limsup_{k\to\infty}
        \frac 1{k}
        \left(
            \sum_{i=1}^k
            \min
            \left(
                \mathcal V^{\mathrm{upper}}_{i\ell_N}
                -
                \mathcal M^{\mathrm{upper}}_{(i-1)\ell_N}
            \right)
        \right)
        \\ &
        =
        \frac 1{\ell_N}
        \mathbb E\left[
            \min\mathcal V^{N|0}_{\ell_N}
        \right]
        % \\ &
        =
        \frac 1{\ell_N}
        \mathbb E\left[
            \mathcal M^{N|0}_{\ell_N}
        \right]
        \\ &
        =
        \frac 1{t\log^3N+o(\log^3N)}
        \left(
            \frac{\pi^2}2\sigma^2t\log N
            +
            o_N(\log N)
        \right)
        \\ &
        =
        \frac{\pi^2}2\sigma^2\frac{1}{\log^2 N}
        +
        o_N\left(\frac{1}{\log^2 N}\right).
    \end{align*}

    \bilingual{
        Similarly, for the lower bound, we construct, by modifying $\mathcal V^{N|1}$ (i.e. an $N$-BRW starting with $N^1=N$ particles at position $0$ on the same probability space such that $\mathcal V^{N|a}$ is a subtree of $\mathcal V^{N|1}$), a lower branching random walk $\mathcal V^{\mathrm{lower}}$ for $\mathcal V^{N|a}$ as follows:
    }{
        类似地,~ 对于下界,~ 通过修改 $\mathcal V^{N|1}$ (即在相同的概率空间上以 $N^1=N$ 个粒子在位置 $0$ 处开始的 $N$-分支随机游走,~ 使得 $\mathcal V^{N|a}$ 是 $\mathcal V^{N|1}$ 的一个子树),~ 构造 $\mathcal V^{N|a}$ 的下界分支随机游走 $\mathcal V^{\mathrm{lower}}$ 如下:
    }
    \begin{itemize}
        \item \bilingual{
            At time $0$, we keep all particles in the original $N$-BRW $\mathcal V^{N|1}$ and let the process evolve as $\mathcal V^{N|1}$ from $0$ till $\ell_N$;
        }{
            在时刻 $0$,~ 保留最初的 $N$-分支随机游走 $\mathcal V^{N|1}$ 中的所有粒子,~ 让过程以 $\mathcal V^{N|1}$ 的方式从 $0$ 演化到 $\ell_N$;
        }
        \item \bilingual{
            At time $\ell_N$, even if the population size $\#\mathcal Z^{N|a}_{\ell_N}$ of $\mathcal V^{N|a}$ does not reach $N$, we translate all the $N$ particles in $\mathcal V^{\mathrm{lower}}_{\ell_N}$ to the left-most position $\min\mathcal V^{\mathrm{lower}}_{\ell_N}$;
        }{
            在时刻 $\ell_N$,~ 即使 $\mathcal V^{N|a}$ 的粒子总数 $\#\mathcal Z^{N|a}_{\ell_N}$ 没有达到 $N$,~ 也将 $\mathcal V^{\mathrm{lower}}_{\ell_N}$ 中的所有 $N$ 个粒子平移到最左侧位置 $\min\mathcal V^{\mathrm{lower}}_{\ell_N}$;
        }
        \item \bilingual{
            From time $\ell_N$ onward, we let the process evolve as $\mathcal V^{N|1}$ and repeat the translation procedure at all times $k\ell_N$ for $k\in\mathbb N$.
        }{
            从时刻 $\ell_N$ 开始,~ 让过程以 $\mathcal V^{N|1}$ 的方式演化,~ 并在所有时刻 $k\ell_N$ 对 $k\in\mathbb N$ 重复平移过程.
        }
    \end{itemize}
    \bilingual{
        By definition of $\mathcal V^{\mathrm{lower}}$, we have, by \Cref{corollary:2} and by induction on $k$, on the same probability space,
    }{
        根据 $\mathcal V^{\mathrm{lower}}$ 的定义有,~ 由\Cref{corollary:2} 和对 $k$ 的归纳法,~ 在相同的概率空间上,
    }
    \begin{equation*}
        \mathcal V^{N|a}_{k\ell_N}
        \succcurlyeq
        \mathcal V^{\mathrm{lower}}_{k\ell_N},
        \quad\forall k\in\mathbb N,
    \end{equation*}
    \bilingual{
        and, by some coupling of $\mathcal V^{\mathrm{lower}}$ and $\mathcal V^{N|a}$,
    }{
        并且,~ 通过 $\mathcal V^{\mathrm{lower}}$ 和 $\mathcal V^{N|a}$ 的某个耦合,
    }
    \begin{equation*}
        \mathcal V^{N|a}_n
        \succcurlyeq
        \mathcal V^{\mathrm{lower}}_n,
        \quad\forall n\in[\![k\ell_N,(k+1)\ell_N]\!],\,\forall k\in\mathbb N.
    \end{equation*}
    \bilingual{
        Hence, there is some coupling of $\mathcal V^{\mathrm{lower}}$ and $\mathcal V^{N|a}$ such that
    }{
        因此,~ 存在 $\mathcal V^{\mathrm{lower}}$ 和 $\mathcal V^{N|a}$ 的某个耦合,~ 使得
    }
    \begin{equation*}
        \mathcal M^{N|a}_n
        \ge
        \mathcal M^{\mathrm{lower}}_n,
        \quad\forall n\in\mathbb N.
    \end{equation*}
    \bilingual{
        Furthermore, by definition of $\mathcal V^{\mathrm{lower}}$, we have
    }{
        此外,~ 根据 $\mathcal V^{\mathrm{lower}}$ 的定义有
    }
    \begin{equation*}
        \mathcal V^{\mathrm{lower}}_{(k+1)\ell_N}
        -
        \mathcal M^{\mathrm{lower}}_{k\ell_N}
        \stackrel{\mathrm{d}}{=}
        \mathcal V^{N|1}_{\ell_N},
        \quad k\in\mathbb N,
    \end{equation*}
    \bilingual{
        and they are independent for different $k$'s. Hence,
    }{
        并且对于不同的 $k$,~ 它们是独立的. ~因此,
    }
    \begin{equation*}
        \left(
            \min\left(
                \mathcal V^{\mathrm{lower}}_{(k+1)\ell_N}
                -
                \mathcal M^{\mathrm{lower}}_{k\ell_N}
            \right):
            k\in\mathbb N
        \right)
    \end{equation*}
    \bilingual{
        is an i.i.d.~sequence of random variables with the same distribution as $\min\mathcal V^{N|1}_{\ell_N}$. To use SLLN, we need to check that, given $N$,
    }{
        是一个独立同分布的随机变量序列,~ 其分布与 $\min\mathcal V^{N|1}_{\ell_N}$ 的分布相同. ~为了使用大数定律,~ 需要检查对于给定的 $N$,
    }
    \begin{equation*}
        \mathbb E\left[
            \min\mathcal V^{N|1}_{\ell_N}
        \right]
        =
        \mathbb E\left[
            \mathcal M^{N|1}_{\ell_N}
        \right]
        <
        \infty.
    \end{equation*}
    \bilingual{
        Indeed, since $\mathcal M^{N|1}_n\le\mathcal M^{N|0}_n$ holds (stocatistically) for all $n$, we have
    }{
        事实上,~ 由于 $\mathcal M^{N|1}_n\le\mathcal M^{N|0}_n$ 对所有 $n$ 都 (随机地) 成立,~ 有
    }
    \begin{equation*}
        \mathbb E\left[
            \left(
                \mathcal M^{N|1}_n
            \right)_+
        \right]
        \le
        \mathbb E\left[
            \left(
                \mathcal M^{N|0}_n
            \right)_+
        \right]
        <
        \infty,
    \end{equation*}
    \bilingual{
        and
    }{
        并且
    }
    \begin{align*}
        \mathbb E\left[
            \left(
                \mathcal M^{N|1}_n
            \right)_-
        \right]
        &
        \le
        \mathbb E\left[
            e^{
                \left(
                    \mathcal M^{N|1}_n
                \right)_-
            }
        \right]
        % \\ &
        \le
        \mathbb E\left[
            e^{
                -\mathcal M^{N|1}_n
            }
            +
            1
        \right]
        \\ &
        \le
        1
        +
        \sum_{|u|=n}
        \mathbb E\left[
            e^{
                -\mathcal V^{N|1}(u)
            }
        \right]
        % \\ &
        =
        1
        +
        N\mathbb E[W_n]
        =
        1+N<\infty,
    \end{align*}
    \bilingual{
        where $W_n$ is the additive martingale, and $\mathbb E[W_n]=1$ is proven in \Cref{lemma:M:exp-neg-moment} under \Cref{assumption:psi-one}, i.e., $\psi(1)=0$. Hence, by SLLN and the proven result of \Cref{proposition:3.1} for $a=1$,
    }{
        其中 $W_n$ 是加性鞅,~ $\mathbb E[W_n]=1$ 在\Cref{assumption:psi-one} 下由\Cref{lemma:M:exp-neg-moment} 证明,~ 即 $\psi(1)=0$. ~因此,~ 通过大数定律和\Cref{proposition:3.1} 已证明的 $a=1$ 的结果,
    }
    \begin{align*}
        v_N
        &=
        \lim_{n\to\infty}
        \frac{1}{n}\mathcal M^{N|a}_n
        =
        \lim_{k\to\infty}
        \frac 1{k\ell_N}
        \mathcal M^{N|a}_{k\ell_N}
        \\ &
        \le
        \liminf_{k\to\infty}
        \frac 1{k\ell_N}
        \mathcal M^{\mathrm{lower}}_{k\ell_N}
        % \\ &
        =
        \frac 1{\ell_N}
        \liminf_{k\to\infty}
        \frac 1{k}
        \left(
            \sum_{i=1}^k
            \left(
                \mathcal M^{\mathrm{lower}}_{i\ell_N}
                -
                \mathcal M^{\mathrm{lower}}_{(i-1)\ell_N}
            \right)
        \right)
        \\ &
        =
        \frac 1{\ell_N}
        \liminf_{k\to\infty}
        \frac 1{k}
        \left(
            \sum_{i=1}^k
            \min
            \left(
                \mathcal V^{\mathrm{lower}}_{i\ell_N}
                -
                \mathcal M^{\mathrm{lower}}_{(i-1)\ell_N}
            \right)
        \right)
        \\ &
        =
        \frac 1{\ell_N}
        \mathbb E\left[
            \min\mathcal V^{N|1}_{\ell_N}
        \right]
        % \\ &
        =
        \frac 1{\ell_N}
        \mathbb E\left[
            \mathcal M^{N|1}_{\ell_N}
        \right]
        \\ &
        =
        \frac 1{t\log^3N+o(\log^3N)}
        \left(
            \left(
                -1+
                \frac{\pi^2}2\sigma^2t
            \right)\log N
            +
            o_N(\log N)
        \right)
        \\ &
        =
        \left(
            -\frac 1t+
            \frac{\pi^2}2\sigma^2
        \right)
        \frac{1}{\log^2 N}
        +
        o_N\left(\frac{1}{\log^2 N}\right),
    \end{align*}
    \bilingual{
        and let $t\to\infty$ to get the desired lower bound. We are now done with the proof of \Cref{theorem:speed}.
    }{
        并让 $t\to\infty$ 来得到所需的下界. ~至此,~ \Cref{theorem:speed} 得证.
    }
\end{proof}
