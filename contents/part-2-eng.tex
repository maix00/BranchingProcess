\begin{lemma}[\bilingual{Small Deviation}{水平管道意义下的小偏差},\ifchinese\hspace{.4em}\else\space\fi A\"id\'ekon--Hu \mycite{aïdékon2025averageweightpercolationcompletegraph}]\label{lemma:4.1}
    \bilingual{
        Let $v>0$ and $0<d<\frac{\pi^2}2\sigma^2$. There exists $\Delta_0>0$ such that for all $\Delta\ge\Delta_0$, $r_0\le 0\le r_0+\Delta$ and $L\ge 1$,
    }{
        设 $v>0$ 且 $0<d<\frac{\pi^2}2\sigma^2$. 存在 $\Delta_0>0$ 使得对所有 $\Delta\ge\Delta_0$, $r_0\le 0\le r_0+\Delta$ 且 $L\ge 1$,
    }
    {
        \setlength{\belowdisplayskip}{0pt}
        \begin{equation}
            \mathbb P(r_0\le Y_j\le r_0+\Delta,\,\forall j\in[\![1,L]\!])\le\exp\left( \Delta^v-\frac{d}{\Delta^2}L\right).
            \label{eq:4.1}
        \end{equation}
    }
\end{lemma}

\begin{proof}\label{proof:lemma:4.1}
    \bilingual{
        By applying \nameref{theorem:mogulskii} (see \Cref{lemma:mogulskii-cal} (i))% and \Cref{example:mogulskii-L-1})
        , for all $a\in(0,1)$ and $\ell=\ell(\Delta)=\lfloor\frac 1d\Delta^{2+v}\rfloor$,
    }{
        通过应用 \nameref{theorem:mogulskii} (参见\Cref{lemma:mogulskii-cal} (i))% 和\Cref{example:mogulskii-L-1})
        , 对所有 $a\in(0,1)$ 和 $\ell=\ell(\Delta)=\lfloor\frac 1d\Delta^{2+v}\rfloor$,
    }
    \begin{equation*}
        \lim_{\Delta\to\infty}
        \frac{\Delta^2}{\ell}
        \log\mathbb P\left(
                -a\Delta\le Y_j\le(1-a)\Delta,\,
                \forall j\in[\![1,\ell]\!]
        \right)
        =
        -\frac{\pi^2}2\sigma^2.
        \label{eq:lemma:4.1:mogulskii-applied}
    \end{equation*}

    \bilingual{
        Fix $m\ge 1$ so that $d{(1+\frac 1m)}^2<\frac{\pi^2}2\sigma^2$. Apply the above equation to $(1+\frac 1m)\Delta$ in lieu of $\Delta$ and
    }{
        固定 $m\ge 1$ 使得 $d{(1+\frac 1m)}^2<\frac{\pi^2}2\sigma^2$. 应用上式, 应用时将上式中的 $\Delta$ 替换为 $(1+\frac 1m)\Delta$, 并且令
    }
    \begin{equation*}
        a=\frac{1-\frac km}{1+\frac 1m},\quad
        k\in[\![0,m-1]\!],
    \end{equation*}
    \bilingual{
        and we can get, for all $k\in[\![0,m-1]\!]$,
    }{
        从而得, 对所有 $k\in[\![0,m-1]\!]$,
    }
    \begin{align*}
        &\quad\lim_{\Delta\to\infty}
        \frac
        {
            {\left(1+\frac 1m\right)}^2
            \Delta^2
        } \ell
        \log\mathbb P\left(
                \left(-1+\frac km\right)\Delta
                \le Y_j
                \le\frac{k+1}{m}\Delta,\,
                \forall j\in[\![1,\ell]\!]
        \right)
        \\ &=
        -\frac{\pi^2}{2}\sigma^2
        <-d{\left(1+\frac 1m\right)}^2.
    \end{align*}
    \bilingual{
        Then we find $\Delta_0=\Delta(d)>0$ such that for all $\Delta\ge\Delta_0$ and $k\in[\![0,m-1]\!]$,
    }{
        于是可得 $\Delta_0=\Delta(d)>0$ 使得对所有 $\Delta\ge\Delta_0$ 和 $k\in[\![0,m-1]\!]$,
    }
    \begin{equation*}
        \mathbb P\left(
            \left(-1+\frac km\right)\Delta
            \le Y_j
            \le\frac{k+1}{m}\Delta,\,
            \forall j\in[\![1,\ell]\!]
        \right)
        \le
        \exp\left(
            -\frac{d}{\Delta^2}\ell
        \right).
    \end{equation*}
    \bilingual{
        Now for $r_0\le 0\le r_0+\Delta$, we can find some $k\in[\![0,m-1]\!]$ such that $\frac{r_0}\Delta+1\in[\frac km,\frac{k+1}m]$, and hence, for all $\Delta\ge\Delta_0$,
    }{
        现在对 $r_0\le 0\le r_0+\Delta$, 存在 $k\in[\![0,m-1]\!]$ 使得 $\frac{r_0}\Delta+1\in[\frac km,\frac{k+1}m]$, 因此对所有 $\Delta\ge\Delta_0$,
    }
    \begin{align*}
        &\quad\mathbb P\left(
                r_0\le Y_j\le r_0+\Delta,\,
                \forall j\in[\![1,\ell]\!]
        \right)
        \\ &\le\mathbb P\left(
            \left(-1+\frac km\right)\Delta
            \le Y_j
            \le\frac{k+1}{m}\Delta,\,
            \forall j\in[\![1,\ell]\!]
        \right)
        \le
        \exp\left(
            -\frac{d}{\Delta^2}\ell
        \right).
    \end{align*}

    \bilingual{
        Now for any $L\ge 1$, split the interval $[\![1,L]\!]$ into $q$ possible intervals of length $\ell$ and one possible interval of length $r$, i.e., $L=q\ell+r$ with $q\ge 0$ and $0\le r<\ell$. Denote in this proof
    }{
        现在对任何 $L\ge 1$, 将区间 $[\![1,L]\!]$ 分成 $q$ 个长度为 $\ell$ 的区间和一个长度为 $r$ 的区间, 即 $L=q\ell+r$, 其中 $q\ge 0$ 且 $0\le r<\ell$. 在本证明中记
    }
    \begin{equation*}
        A_k
        :=
        \{
            r_0\le Y_j\le r_0+\Delta,\,
            \forall j\in[\![(k-1)\ell+1,k\ell]\!]
        \},
        \quad
        k\in[\![1,q]\!].
    \end{equation*}
    \bilingual{
        Write $B_0:=\Omega$ and $B_k:=\bigcap_{l=1}^k A_l$ for $1\le k\le q$.
    }{
        记 $B_0:=\Omega$ 且 $B_k:=\bigcap_{l=1}^k A_l$ 对 $1\le k\le q$.
    }
    \bilingual{
        By the Markov property of $(Y_n)$, for each $1\le k\le q$,
    }{
        由随机游走 $(Y_n)$ 的 Markov 性质, 对每个 $1\le k\le q$,
    }
    \begin{equation*}
        \mathbb E\left({\bf 1}_{A_k}\middle|\mathscr F_{(k-1)\ell}\right)
        =p\!\left(Y_{(k-1)\ell}\right),
    \end{equation*}
    \bilingual{
        where
    }{
        其中
    }
    \begin{align*}
        p(x)&:=\mathbb P\left(
            r_0-x\le Y_j\le r_0+\Delta-x,
            \,\forall j\in[\![1,\ell]\!]
        \right),\quad x\in\mathbb R,\\
        0\le p(x)&\le\exp\left(-\frac{d\ell}{\Delta^2}\right),
        \quad x\in[r_0,r_0+\Delta].
    \end{align*}
    \bilingual{
        Here the one-block estimate applies since
        $r_0-x\in[-\Delta,0]$ and $r_0+\Delta-x\in[0,\Delta]$.
    }{
        此处不等式使用单块估计, 因为对每个
        $x\in[r_0,r_0+\Delta]$, 均有
        $r_0-x\in[-\Delta,0]$ 且 $r_0+\Delta-x\in[0,\Delta]$.
    }
    \bilingual{
        On $B_{k-1}$, the block starts in $[r_0,r_0+\Delta]$; for $k=1$ this
        follows from $Y_0=0$ and $r_0\le0\le r_0+\Delta$. Thus the tower
        property gives the following recursion:
    }{
        在事件 $B_{k-1}$ 上, 分块起点满足
        $Y_{(k-1)\ell}\in[r_0,r_0+\Delta]$. 当 $k=1$ 时, 该关系由
        $Y_0=0$ 和 $r_0\le0\le r_0+\Delta$ 得到; 当 $2\le k\le q$ 时,
        $B_{k-1}\subseteq A_{k-1}$, 故由 $A_{k-1}$ 的定义得到该关系.
        因而由塔式性质得到下列递推式:
    }
    \begin{equation*}
        \mathbb P(B_k)
        =\mathbb E\left[{\bf 1}_{B_{k-1}}p\!\left(Y_{(k-1)\ell}\right)\right]
        \le\exp\left(-\frac{d\ell}{\Delta^2}\right)\mathbb P(B_{k-1}),
        \quad 1\le k\le q.
    \end{equation*}
    \bilingual{
        Iterating this recursion and using $L=q\ell+r$ gives
    }{
        迭代上述递推式并利用 $L=q\ell+r$, 得
    }
    \begin{equation*}
        \mathbb P(B_q)
        \le\exp\left(-\frac{dq\ell}{\Delta^2}\right)
        =\exp\left(\frac{dr}{\Delta^2}-\frac{dL}{\Delta^2}\right)
        \le\exp\left(\Delta^v-\frac{dL}{\Delta^2}\right).
    \end{equation*}
    \bilingual{
        The event up to time $L$ is contained in $B_q$, and the last
        inequality uses $r<\ell\le\frac{1}{d}\Delta^{2+v}$.
    }{
        截至时刻 $L$ 的事件包含于 $B_q$, 最后一式由
        $r<\ell\le\frac{1}{d}\Delta^{2+v}$ 得到.
    }
    \bilingual{
        This proves the claim.
    }{
        此即得证.
    }
\end{proof}

\begin{lemma}[\bilingual{Small Deviation under No Large Drop}{无大回撤意义下的小偏差},\ifchinese\hspace{.4em}\else\space\fi A\"id\'ekon--Hu \mycite{aïdékon2025averageweightpercolationcompletegraph}]
    \label{lemma:2.3}
    \bilingual{
        Let $c>0$. For any $d\in(0,\frac{\pi^2}2\sigma^2)$, there exists $\Delta_0=\Delta_0(c,d)\ge 1$ such that for all $\Delta\ge\Delta_0$ and $L\ge\Delta^{2+c}$, we have
    }{
        设 $c>0$. 对任何 $d\in(0,\frac{\pi^2}2\sigma^2)$, 存在 $\Delta_0=\Delta_0(c,d)\ge 1$ 使得对所有 $\Delta\ge\Delta_0$ 和 $L\ge\Delta^{2+c}$, 成立
    }
    {
        \setlength\belowdisplayskip{-20pt}
        \begin{equation*}
            \mathbb P\left(
                Y_L
                \le
                \frac\Delta c,\,
                \min_{0\le i\le j\le L}
                (Y_j-Y_i)
                \ge
                -\Delta
            \right)
            \le
            e^{-\frac{dL}{\Delta^2}}.
        \end{equation*}
    }
\end{lemma}

\begin{proof}
    \label{proof:lemma:2.3}\quad

    \bilingual{
        We denote
    }{
        记事件
    }
    \begin{equation*}
        B = B(c,\Delta,L) := \left\{ Y_L \le \frac\Delta c, \, \min_{0\le i\le j\le L} (Y_j-Y_i) \ge -\Delta \right\}.
    \end{equation*}
    \bilingual{
        Note that $B(c_1,\Delta,L) \subset B(c_2,\Delta,L)$ for all $c_1 \ge c_2$ and fixed $\Delta, L$.
    }{
        注意到对于固定的 $\Delta$ 和 $L$, 若 $c_1 \ge c_2$, 则有 $B(c_1,\Delta,L) \subset B(c_2,\Delta,L)$.
    }

    \bilingual{
        For the case $c \ge \frac{1}{2}$, it suffices to prove the result for $c' \in (0, \frac{1}{2})$ and $L \ge \Delta^{\frac{2}{1-2c'}}$. Indeed, for any $c \ge \frac{1}{2}$, there exists $c'$ such that
    }{
        对于 $c \ge \frac{1}{2}$ 的情形, 只需证明原结论对 $c' \in (0, \frac{1}{2})$ 且 $L \ge \Delta^{\frac{2}{1-2c'}}$ 成立. 事实上, 对任何 $c \ge \frac{1}{2}$, 总能选取 $c'$ 满足
    }
    \begin{equation*}
        0 < \frac{1}{2} - \frac{1}{2+c} \le c' < \frac{1}{2} \le c.
    \end{equation*}
    \bilingual{
        Applying the result for $c'$ and $L \ge \Delta^{\frac{2}{1-2c'}}$, there exists $\Delta_1 = \Delta_1(c',d) = \Delta_1(c,d) \ge 1$ such that for all $\Delta \ge \Delta_1$ and $L \ge \Delta^{\frac{2}{1-2c'}}$
    }{
        对此 $c'$ 和 $L \ge \Delta^{\frac{2}{1-2c'}}$ 应用结论, 存在 $\Delta_1 = \Delta_1(c',d) = \Delta_1(c,d) \ge 1$ 使得对所有 $\Delta \ge \Delta_1$ 且 $L \ge \Delta^{\frac{2}{1-2c'}}$ 成立
    }
    \begin{equation*}
        \mathbb{P}(B(c,\Delta,L)) \le \mathbb{P}(B(c',\Delta,L)) \le e^{-\frac{dL}{\Delta^2}}.
    \end{equation*}
    \bilingual{
        Since $\Delta^{\frac{2}{1-2c'}} \le \Delta^{2+c}$, the above inequality holds for all $L \ge \Delta^{2+c}$. In this regime, for $0 < d < \frac{\pi^2}{2}\sigma^2$, we choose $\eta \in (0,1)$ sufficiently small such that
    }{
        由于 $\Delta^{\frac{2}{1-2c'}} \le \Delta^{2+c}$, 上式对所有 $L \ge \Delta^{2+c}$ 均成立. 此时, 对于 $0 < d < \frac{\pi^2}{2}\sigma^2$, 选取足够小的 $\eta \in (0,1)$ 使得
    }
    \begin{equation*}
        (2+\eta)\left(\frac{1}{2}-c\right) < 1, \quad d(1+2\eta)^4 < \frac{\pi^2}{2}\sigma^2.
    \end{equation*}

    \bilingual{
        For $0 < c < \frac{1}{2}$, we directly prove the result for $L \ge \Delta^{2+c}$ using the same method, only requiring $\eta \in (0,1)$ to be small enough to satisfy $\eta < c$ and the second inequality above.
    }{
        对于 $0 < c < \frac{1}{2}$ 的情形, 直接用相同方法证明 $L \ge \Delta^{2+c}$ 时的结论, 区别仅在于需令 $\eta \in (0,1)$ 足够小以满足 $\eta < c$ 且成立上述第二个不等式.
    }

    \bilingual{
        Note that in both cases, we have
    }{
        注意到, 无论上述哪种情况, 均满足
    }
    \begin{equation*}
        \frac{1}{c} \ge 2, \quad L > \lfloor\Delta^{2+\eta}\rfloor, \quad \lim_{\Delta\to\infty} \frac{L}{\lfloor\Delta^{2+\eta}\rfloor} = \infty.
    \end{equation*}
    \bilingual{
        We divide the interval $[\![0,L]\!]$ into smaller intervals of length $\lfloor\Delta^{2+\eta}\rfloor$, and the space into intervals of length $\lceil\eta\Delta\rceil$. Let $q := \lfloor\frac{L}{\lfloor\Delta^{2+\eta}\rfloor}\rfloor$, $\ell_0 := 0$, and
    }{

    }{

    }{

    }{

    \end{equation*}

        which completes the proof.
    }{
        证毕.

\end{proof}

            &(\widehat v_{2,n}-a_l)d_n&=(b_l+v_2)\Delta_n(k_n).
        \end{aligned}
    \end{equation*}
    \bilingual{
        Thus the event in part (ii) is exactly $\mathcal Q_{\Delta_n,l_n,k_n}$, and
    }{
        因而第 (ii) 部分中的事件恰为 $\mathcal Q_{\Delta_n,l_n,k_n}$, 且
    }
    \begin{equation*}
        \frac{d_n^2}{m_n}\log\mathbb P(\mathcal Q_{\Delta_n,l_n,k_n})
        \longrightarrow-\frac{\pi^2}{2}\sigma^2.
    \end{equation*}
    \bilingual{
        Because $d_n^2=w^2\Delta_n(k_n)^2$ and $\Delta_n(k_n)/\Delta_n\to1$, this gives $R_{\Delta_n,l_n,k_n}\to0$, contrary to $\varepsilon\le|R_{\Delta_n,l_n,k_n}|$. Hence for each $\varepsilon>0$ the set of such triples has bounded $\Delta$-coordinates, which is the claimed uniform limit. The reflected terminal window in \eqref{eq:4.3:2} has the same proof, with its fixed interior interval and reflected endpoint fractions.
    }{
        由于 $d_n^2=w^2\Delta_n(k_n)^2$ 且 $\Delta_n(k_n)/\Delta_n\to1$, 上式给出 $R_{\Delta_n,l_n,k_n}\to0$, 与 $\varepsilon\le|R_{\Delta_n,l_n,k_n}|$ 矛盾. 因此, 对每个 $\varepsilon>0$, 满足该不等式的三元组的 $\Delta$ 坐标有界, 这正是所需的一致极限. 对 \eqref{eq:4.3:2} 中反射后的终端窗, 使用其固定内部区间及反射后的端点比例可作同样证明.
    }
    \bilingual{
        Since there are finitely many grid cells and $d''w^2>\frac{\pi^2}2\sigma^2$, the resulting lower bound holds for all sufficiently large $\Delta$ uniformly over $l$ and the block choices:
    }{
        由于网格单元只有有限个且 $d''w^2>\frac{\pi^2}2\sigma^2$, 对充分大的 $\Delta$, 下界对所有 $l$ 及各分块选择一致成立:
    }
    \begin{equation*}
        \mathbb P(\mathcal Q_{\Delta,l,k})
        \ge
        \exp\left(-\frac{d''}{\Delta^2}|I_k|\right).
    \end{equation*}
    \bilingual{
        For each $r'_0\in[-\frac23\Delta(k),-\frac13\Delta(k)]$, choose $l$ such that $r'_0/\Delta(k)\in[b_l,b_{l+1}]$. The common narrowed-tube event is contained in the desired event for this $r'_0$, so monotonicity of probability gives
    }{
        对每个 $r'_0\in[-\frac23\Delta(k),-\frac13\Delta(k)]$, 取 $l$ 使 $r'_0/\Delta(k)\in[b_l,b_{l+1}]$. 共同缩窄管道事件包含于该 $r'_0$ 对应的目标事件, 因而由概率单调性得到
    }
    \begin{equation*}
        \mathbb P\left(\;
            \begin{aligned}
                r'_0\le \tilde Y_j\le r'_0+\Delta(k),\,
                \forall j,\\
                r'_0+v_1\Delta(k)\le \tilde Y_{|I_k|},\\
                \tilde Y_{|I_k|}\le r'_0+v_2\Delta(k)
            \end{aligned}
        \;\right)
        \ge
        \mathbb P\left(
            \begin{aligned}
                b_{l+1}\le\frac{\tilde Y_j}{\Delta(k)}\le b_l+1,\quad&\forall j,\\
                b_{l+1}+v_1\le\frac{\tilde Y_{|I_k|}}{\Delta(k)}\le b_l+v_2
            \end{aligned}
        \right)
        \ge
        e^{
            -\frac{d''}{\Delta^2}|I_k|
        },
    \end{equation*}
    \bilingual{
        Finally, using the Markov property in the reverse order, the probability of the event $E$ in \eqref{eq:4.3} is bounded below by the product of probabilities, which yields, for $\Delta\ge\Delta_0$,
    }{
        最后, 倒序利用 Markov 性质, \eqref{eq:4.3} 中事件 $E$ 的概率被各分块概率之积所下界控制, 从而得到, 在 $\Delta\ge\Delta_0$ 时,
    }
    \begin{align*}
        \mathbb{P}(E)
        &\ge
        \mathbb E\left[
            \mathbf 1_{E_0} \prod_{k=1}^{q} \mathbf 1_{E_k}
        \right]
        =
        \mathbb E\left[
            \mathbf 1_{E_0} \prod_{k=1}^{q-1} \mathbf 1_{E_k}
            \mathbb P\left(E_q \middle| \mathcal F_{t_q}\right)
        \right]
        \\
        &
        \ge \mathbb{P}(E_0) \prod_{k=1}^{q} e^{-\frac{d''}{\Delta^2}|I_k|}
        \ge e^{-\frac{d' - d''}{\Delta^2}L} e^{-\frac{d''}{\Delta^2}(L-t_1)}
        \ge e^{-\frac{d'}{\Delta^2}L},
    \end{align*}
    \bilingual{
        which completes the proof of \eqref{eq:4.3}.
    }{
        此即证得 \eqref{eq:4.3}.
    }
    \qedhere
\end{proof}
