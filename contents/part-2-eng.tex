\begin{lemma}[\bilingual{Small Deviation}{水平管道意义下的小偏差}, A\"id\'ekon--Hu \mycite{aïdékon2025averageweightpercolationcompletegraph}]\label{lemma:4.1}
    \bilingual{
        Let $v>0$ and $0<d<\frac{\pi^2}2\sigma^2$. There exsits $\Delta_0>0$ such that for all $\Delta\ge\Delta_0$, $r_0\le 0\le r_0+\Delta$ and $\ell\ge 1$,
    }{
        设 $v>0$ 且 $0<d<\frac{\pi^2}2\sigma^2$. ~存在 $\Delta_0>0$ 使得对所有 $\Delta\ge\Delta_0$,~ $r_0\le 0\le r_0+\Delta$ 且 $\ell\ge 1$,
    }
    {
        \setlength{\belowdisplayskip}{0pt}
        \begin{equation}
            \mathbb P(r_0\le Y_j\le r_0+\Delta,\,\forall j\in[\![1,\ell]\!])\le\exp\left( \Delta^v-\frac{d}{\Delta^2}\ell\right).
            \label{eq:4.1}
        \end{equation}
    }
\end{lemma}

\begin{proof}[\bilingual{Proof of \Cref{lemma:4.1}}{\Cref{lemma:4.1}~的证明}]\label{proof:lemma:4.1}
    \bilingual{
        By applying \nameref{theorem:mogulskii} (see \Cref{lemma:mogulskii-cal} (iii) and \Cref{example:mogulskii-L-1}), for all $a\in(0,1)$ and $L=L(\Delta)=\lfloor\frac 1d\Delta^{2+v}\rfloor$,
    }{
        通过应用~\nameref{theorem:mogulskii}~(参见\Cref{lemma:mogulskii-cal} (iii) 和\Cref{example:mogulskii-L-1}),~ 对所有 $a\in(0,1)$ 和 $L=L(\Delta)=\lfloor\frac 1d\Delta^{2+v}\rfloor$,~ 
    }
    \begin{equation*}
        \lim_{\Delta\to\infty}
        \frac{\Delta^2}{L}
        \log\mathbb P\left(
                -a\Delta\le Y_j\le(1-a)\Delta,\,
                \forall j\in[\![1,L]\!]
        \right)
        =
        -\frac{\pi^2}2\sigma^2.
        \label{eq:lemma:4.1:mogulskii-applied}
    \end{equation*}

    \bilingual{
        Fix $m\ge 1$ so that $d{(1+\frac 1m)}^2<\frac{\pi^2}2\sigma^2$. Apply the above equation to $(1+\frac 1m)\Delta$ in lieu of $\Delta$ and
    }{
        固定 $m\ge 1$ 使得 $d{(1+\frac 1m)}^2<\frac{\pi^2}2\sigma^2$. ~应用上式,~ 应用时将上式中的 $\Delta$ 替换为 $(1+\frac 1m)\Delta$,~ 并且令
    }
    \begin{equation*}
        a=\frac{1-\frac km}{1+\frac 1m},\quad
        k\in[\![0,m-1]\!],
    \end{equation*}
    \bilingual{
        and we can get, for all $k\in[\![0,m-1]\!]$,
    }{
        从而得,~ 对所有 $k\in[\![0,m-1]\!]$,~ 
    }
    \begin{align*}
        &\quad\lim_{\Delta\to\infty}
        \frac 
        {
            {\left(1+\frac 1m\right)}^2
            \Delta^2
        } L
        \log\mathbb P\left(
                \left(-1+\frac km\right)\Delta
                \le Y_j
                \le\frac{k+1}{m}\Delta,\,
                \forall j\in[\![1,L]\!]
        \right)
        \\ &=
        -\frac{\pi^2}{2}\sigma^2
        <-d{\left(1+\frac 1m\right)}^2.
    \end{align*}
    \bilingual{
        Then we find $\Delta_0=\Delta(d)>0$ such that for all $\Delta\ge\Delta_0$ and $k\in[\![0,m-1]\!]$,
    }{
        于是可得 $\Delta_0=\Delta(d)>0$ 使得对所有 $\Delta\ge\Delta_0$ 和 $k\in[\![0,m-1]\!]$,~ 
    }
    \begin{equation*}
        \mathbb P\left(
            \left(-1+\frac km\right)\Delta
            \le Y_j
            \le\frac{k+1}{m}\Delta,\,
            \forall j\in[\![1,L]\!]
        \right)
        \le
        \exp\left(
            -\frac{d}{\Delta^2}L
        \right).
    \end{equation*}
    \bilingual{
        Now for $r_0\le 0\le r_0+\Delta$, we can find some $k\in[\![0,m-1]\!]$ such that $\frac{r_0}\Delta+1\in[\frac km,\frac{k+1}m]$, and hence, for all $\Delta\ge\Delta_0$,
    }{
        现在对 $r_0\le 0\le r_0+\Delta$,~ 存在 $k\in[\![0,m-1]\!]$ 使得 $\frac{r_0}\Delta+1\in[\frac km,\frac{k+1}m]$,~ 因此对所有 $\Delta\ge\Delta_0$,~ 
    }
    \begin{align*}
        &\quad\mathbb P\left(
                r_0\le Y_j\le r_0+\Delta,\,
                \forall j\in[\![1,L]\!]
        \right)
        \\ &\le\mathbb P\left(
            \left(-1+\frac km\right)\Delta
            \le Y_j
            \le\frac{k+1}{m}\Delta,\,
            \forall j\in[\![1,L]\!]
        \right)
        \le
        \exp\left(
            -\frac{d}{\Delta^2}L
        \right).
    \end{align*}
    
    \bilingual{
        Now for any $\ell\ge 1$, split the interval $[\![1,\ell]\!]$ into $q$ possible intervals of length $L$ and one possible interval of length $r$, i.e., $\ell=qL+r$ with $q\ge 0$ and $0\le r<L$. Denote in this proof
    }{
        现在对任何 $\ell\ge 1$,~ 将区间 $[\![1,\ell]\!]$ 分成 $q$ 个长度为 $L$ 的区间和一个长度为 $r$ 的区间,~ 即 $\ell=qL+r$,~ 其中 $q\ge 0$ 且 $0\le r<L$. ~在本证明中记
    }
    \begin{equation*}
        A_k
        :=
        \{
            r_0\le Y_j\le r_0+\Delta,\,
            \forall j\in[\![(k-1)L+1,kL]\!]
        \},
        \quad
        k\in[\![1,q]\!].
    \end{equation*}
    \bilingual{
        We have
    }{
        从而
    }
    \begin{align*}
        &\quad
        \mathbb P(
            r_0\le Y_j\le r_0+\Delta,\,
            \forall j\in[1,\ell]
        )
        \\ &
        \le
        \mathbb P(
            \cap_{k=1}^qA_k
        )
        =
        \prod_{k=1}^q\mathbb P\left(
            A_k
            \middle|
            \cap_{l=1}^{k-1}A_l
        \right)
        \\ &
        =
        \prod_{k=1}^q\left(
            \frac 1{
                \mathbb P\left(
                    \cap_{l=1}^{k-1}A_l
                \right)
            }
            \mathbb E\left[
                \mathbb E\left(
                    {\bf 1}_{A_k}
                    \middle|
                    \mathscr F_{(k-1)L}
                \right)
                {\bf 1}_{
                    \cap_{l=1}^{k-1}A_l
                }
            \right]
        \right)
        \\ &
        \le
        \prod_{k=1}^q
        \sup_{x\in r_0+[0,\Delta]}
        \mathbb P(
            r_0\le Y_j+x\le r_0+\Delta,\,
            \forall j\in[\![1,L]\!]
        )
        \\ &
        \le
        \left[
            \exp\left(
                -\frac d{\Delta^2}L
            \right)
        \right]^q
        =
        \exp\left(\frac{d}{\Delta^2}r-\frac{d}{\Delta^2}\ell\right)
        % \\ &
        \le
        \exp\left(\Delta^v-\frac{d}{\Delta^2}\ell\right),
    \end{align*}
    \bilingual{
        where $r_0-x\in[-\Delta,0]$ for $x\in r_0+[0,\Delta]$, and the last inequality follows from $r<L\le\frac{1}{d}\Delta^{2+v}$.
    }{
        其中 $r_0-x\in[-\Delta,0]$ 对 $x\in r_0+[0,\Delta]$ 成立,~ 并且最后一个不等式是因为 $r<L\le\frac{1}{d}\Delta^{2+v}$.
    }
\end{proof}

\begin{lemma}[\bilingual{Small Deviation with Small Drift}{小漂移倾斜管道意义下的小偏差}]\label{lemma:4.2}
    \bilingual{
        Let $v>0$, $0<d<\frac{\pi^2}2\sigma^2$ and $c\ge 0$. There exsits $\Delta_0>0$ such that for all $\Delta\ge\Delta_0$, $r_0\le 0\le r_0+\Delta$ and $\ell\ge 1$,
    }{
        设 $v>0$,~ $0<d<\frac{\pi^2}2\sigma^2$ 且 $c\ge 0$. ~存在 $\Delta_0>0$ 使得对所有 $\Delta\ge\Delta_0$,~ $r_0\le 0\le r_0+\Delta$ 且 $\ell\ge 1$,
    }
    \begin{equation}
        \mathbb P\left(
            r_j\le Y_j\le s_j,\,
            \forall j\in[\![1,\ell]\!]
        \right)
        \le
        \exp\left(
            \Delta^v-\frac d{\Delta^2}\ell
        \right),
        \label{eq:4.2}
    \end{equation}
    \bilingual{
        where $r_j:=r_0+\frac c{\Delta^2}j$ and $s_j:=r_j+\Delta$.
    }{
        其中 $r_j:=r_0+\frac c{\Delta^2}j$ 且 $s_j:=r_j+\Delta$.
    }
\end{lemma}

\begin{proof}[\bilingual{Proof of \Cref{lemma:4.2}}{\Cref{lemma:4.2} 的证明}]\label{proof:lemma:4.2}
    \quad

    \bilingual{
        We cancel the drift by Exponential Change of Measure (ECM, or Esscher tilting).
    }{
        通过指数测度变换来消除漂移.
    }
    
    \bilingual{
        Denote $q:=\frac c{\Delta^2}\ge 0$. Construct a measure $\mathbb P_\theta$ from the Radon--Nikodym derivative
    }{
        记 $q:=\frac c{\Delta^2}\ge 0$. ~从如下 Radon--Nikodym 导数构造测度 $\mathbb P_\theta$ 
    }
    \begin{equation}
        \frac{\mathrm d{(\mathbb P_\theta)}_\xi}{\mathrm d\mathbb P_\xi}
        (x)
        =
        \exp\left(
            \theta x-\kappa(\theta)
        \right),
        \quad
        x\in\mathbb R
        \label{eq:ecm:rn-derivative}
    \end{equation}
    \bilingual{
        for some $\theta=\theta(q)\in\mathbb R$ to be specified later such that $\mathbb E_\theta\xi=q$, where $\kappa(\theta):=\log \mathbb Ee^{\theta\xi}$ is the log-Laplace transform of the step $\xi$. Note that on the interior of the domain of $\kappa$, by the property of $\kappa$ (see \Cref{lemma:kappa}), we have
    }{
        其中 $\theta=\theta(q)\in\mathbb R$ 将在后面指定,~ 使得 $\mathbb E_\theta\xi=q$,~ 这里 $\kappa(\theta):=\log \mathbb Ee^{\theta\xi}$ 是步长 $\xi$ 的 log-Laplace 变换. ~注意在 $\kappa$ 的定义域的内部,~ 根据 $\kappa$ 的性质 (参见\Cref{lemma:kappa}) 有
    }
    \begin{align*}
        \mathbb E_\theta\xi
        &=
        \mathbb E\left[
            \xi\frac{\mathrm d{(\mathbb P_\theta)}_\xi}{\mathrm d\mathbb P_\xi}(\xi)
        \right]
        =
        \frac{\mathbb E\left[\xi e^{\theta\xi}\right]}{\mathbb E\left[e^{\theta\xi}\right]}
        =\kappa'(\theta),
        \\
        \mathbb E_\theta\xi^2
        &=
        \mathbb E\left[
            \xi^2\frac{\mathrm d{(\mathbb P_\theta)}_\xi}{\mathrm d\mathbb P_\xi}(\xi)
        \right]
        =
        \frac{\mathbb E\left[\xi^2 e^{\theta\xi}\right]}{\mathbb E\left[e^{\theta\xi}\right]}
        =\kappa''{(\theta)+(\kappa'(\theta))}^2.
    \end{align*}
    \bilingual{
        Note that $\kappa''(\theta)=\mathrm{Var}_\theta\xi>0$ since $\xi$ is not degenerate.
    }{
        注意$\kappa''(\theta)=\mathrm{Var}_\theta\xi>0$,~ 此因$\xi$不是退化的.
    }
    
    \bilingual{
        It is obvious that $\kappa(0)=0$. By the relation $\kappa(\theta)=\psi(1-\theta)$ (see \Cref{lemma:kappa}), we know that $\kappa'(0)=0$ (since $\psi'(1)=0$, \Cref{assumption:psi-prime-one}), that $\kappa''(0)=\psi''(1)=\sigma^2<\infty$, and that $\kappa$ is analytic at $0$ (see \Cref{lemma:kappa}), so we can write
    }{
        显然 $\kappa(0)=0$. ~根据关系 $\kappa(\theta)=\psi(1-\theta)$ (参见\Cref{lemma:kappa}) 可知 $\kappa'(0)=0$ (此因 $\psi'(1)=0$,~ \Cref{assumption:psi-prime-one}),~ $\kappa''(0)=\psi''(1)=\sigma^2<\infty$,~ 并且 $\kappa$ 在 $0$ 处解析（参见\Cref{lemma:kappa}）,~ 因此可写出
    }
    \begin{align*}
        \kappa(\theta)
        &=
        \frac{\sigma^2}{2}\theta^2
        +
        \frac{\kappa^{(3)}(0)}{6}\theta^3
        +
        O_\theta(\theta^4),
        \\
        \kappa'(\theta)
        &=
        \sigma^2\theta
        +
        \frac{\kappa^{(3)}(0)}{2}\theta^2
        +
        O_\theta(\theta^3),
        \\
        \kappa''(\theta)
        &=
        \sigma^2
        +
        \kappa^{(3)}(0)\theta
        +
        O_\theta(\theta^2).
    \end{align*}
    \bilingual{
        Let $\mathbb E_\theta\xi=\kappa'(\theta)=q$ and we can get an analytic function $\theta=\theta(q)$ by Inverse Function Theorem as 
    }{
        令 $\mathbb E_\theta\xi=\kappa'(\theta)=q$,~ 可通过反函数定理得到解析函数 $\theta=\theta(q)$,~ 表示为
    }
    \begin{equation}
        \theta(q)
        =
        \frac{1}{\sigma^2}q+
        O_q(q^2).
        \label{eq:ecm:theta}
    \end{equation}
    \bilingual{
        Since $\kappa''> 0$, $\kappa'(0)=0$, $\kappa'(\theta)=q\ge 0$, we have $\theta(q)\ge 0$.
    }{
        由于 $\kappa''> 0$,~ $\kappa'(0)=0$,~ $\kappa'(\theta)=q\ge 0$,~ 有 $\theta(q)\ge 0$.
    }

    \bilingual{
        Denote $\widetilde Y_j:=Y_j-qj$, we have
    }{
        记 $\widetilde Y_j:=Y_j-qj$,~ 有
    }
    \begin{align*}
        % &\quad
        \mathbb P\left(
            r_j\le Y_j\le s_j,\,
            \forall j\in[\![1,\ell]\!]
        \right)
        % \\
        &
        =
        \mathbb E_\theta\left[
            \left(
                \prod_{i=1}^\ell
                \frac{\mathrm d\mathbb P_\xi}{\mathrm d{(\mathbb P_\theta)}_\xi}(\xi_i)
            \right)
            {\bf 1}_{\left\{
                r_j\le Y_j\le s_j+\Delta,\,
                \forall j\in[\![1,\ell]\!]
            \right\}}
        \right]
        \\ &=
        e^{\ell\kappa(\theta)}
        \mathbb E_\theta\left[
            e^{-\theta Y_\ell}
            {\bf 1}_{\left\{
                r_j\le Y_j\le s_j,\,
                \forall j\in[\![1,\ell]\!]
            \right\}}
        \right]
        \\ &=
        e^{\ell(\kappa(\theta)-\theta q)}
        \mathbb E_\theta\left[
            e^{-\theta\widetilde Y_\ell}
            {\bf 1}_{\left\{
                r_0\le\widetilde Y_j\le r_0+\Delta,\,
                \forall j\in[\![1,\ell]\!]
            \right\}}
        \right]
        \\ &\le
        e^{\ell(\kappa(\theta)-\theta q)}
        \mathbb E_\theta\left[
            e^{-\theta r_0}
            {\bf 1}_{\left\{
                r_0\le\widetilde Y_j\le r_0+\Delta,\,
                \forall j\in[\![1,\ell]\!]
            \right\}}
        \right].
    \end{align*}
    \bilingual{
        We break this last expression into three parts:
    }{
        将最后的表达式分为三部分：
    }
    \begin{equation*}
        e^{\ell(\kappa(\theta)-\theta q)},\,
        e^{-\theta r_0},\,
        \mathbb P_\theta\left(
            r_0\le\widetilde Y_j\le r_0+\Delta,\,
            \forall j\in[\![1,\ell]\!]
        \right).
    \end{equation*}
    
    \bilingual{
        Consider the third part first. Note that $(\widetilde Y_j)$ is a random walk with step distribution $\xi-q=\xi-\mathbb E_\theta\xi$, so the step is of zero mean and finite variance $\mathbb E_\theta[(\xi-q)^2]=\kappa''(\theta)$. Since $\kappa$ is analytic at $0$ and $\theta\to 0$ as $\Delta\to\infty$, we have $\kappa''(\theta)\to\kappa''(0)=\sigma^2$ as $\Delta\to\infty$.     }{
