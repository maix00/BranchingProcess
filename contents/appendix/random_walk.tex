\section{\bilingual{Random-Walk Lemmas}{随机游走引理}}
\label{section:random-walk-lemmas}

\begin{lemma}[\bilingual{Meander estimate for the transition event}{基于 Brownian 蜿蜒的过渡事件估计}]
    \label{lemma:meander-transition}
    \bilingual{
        Let $(\xi_j)_{j\ge1}$ be i.i.d. random variables with mean $0$ and variance $\sigma^2\in(0,\infty)$. Fix $c\ge0$ and $\alpha>0$. For $\Delta>0$, set $n_\Delta:=\lfloor\Delta^2\rfloor$ and $h_j:=cj/\Delta^2$. For $x\in[0,\Delta]$, let $S_0=x$, $S_j:=x+\sum_{i=1}^j\xi_i$, and define
    }{
        设 $(\xi_j)_{j\ge1}$ 为均值 $0$、方差 $\sigma^2\in(0,\infty)$ 的独立同分布随机变量. 固定 $c\ge0$ 与 $\alpha>0$. 对 $\Delta>0$, 令 $n_\Delta:=\lfloor\Delta^2\rfloor$ 且 $h_j:=cj/\Delta^2$. 对 $x\in[0,\Delta]$, 令 $S_0=x$, $S_j:=x+\sum_{i=1}^j\xi_i$, 并定义
    }
    \begin{equation*}
        \mathcal A_\Delta(x):=
        \left\{
            h_j\le S_j\le\Delta+h_j\ (1\le j\le n_\Delta),\quad
            S_{n_\Delta}\in\left[h_{n_\Delta}+\frac13\Delta,
                h_{n_\Delta}+\frac23\Delta\right]
        \right\}.
    \end{equation*}
    \bilingual{Then there are $C_\alpha>0$ and $\Delta_0>0$ such that}{则存在 $C_\alpha>0$ 与 $\Delta_0>0$, 使得}
    \begin{equation*}
        \inf_{x\in[0,\Delta]}\mathbb P_x\bigl(\mathcal A_\Delta(x)\bigr)
        \ge C_\alpha\Delta^{-1-\alpha},
        \qquad \Delta\ge\Delta_0,
    \end{equation*}
    \bilingual{where $\mathbb P_x$ denotes the law with $S_0=x$.}{其中 $\mathbb P_x$ 表示 $S_0=x$ 时的概率律.}
\end{lemma}

\begin{proof}[\bilingual{Proof of \Cref{lemma:meander-transition}}{\Cref{lemma:meander-transition} 的证明}]
    \bilingual{
        For $\theta\in\{-1,1\}$, let $Z_0^\theta=0$ and let the increments of $(Z_j^\theta)$ have the law of $\theta\xi_1$. Define
    }{
        对 $\theta\in\{-1,1\}$, 令 $Z_0^\theta=0$, 且 $(Z_j^\theta)$ 的增量服从 $\theta\xi_1$ 的分布. 定义
    }
    \begin{equation*}
        \begin{aligned}
            C_N^\theta&:=\{Z_j^\theta\ge0,\ 1\le j\le N\},\\
            D_{N,\Delta}^\theta&:=
            \left\{
                \begin{aligned}
                    0&\le Z_j^\theta\le\frac56\Delta,\quad 1\le j\le N,\\
                    Z_N^\theta&\in\left[\frac5{12}\Delta,\frac{11}{24}\Delta\right]
                \end{aligned}
            \right\}
        \end{aligned}
    \end{equation*}
    \bilingual{
        In the Skorokhod space $D([0,1])$ of càdlàg paths, the conditioned invariance principle of Bolthausen~\mycite{Bolthausen_1976}, Theorem 3.2, together with the meander identification in Iglehart~\mycite{Iglehart_1974}, Section 3, and Caravenna and Chaumont~\mycite{Caravenna_Chaumont_2008}, equation (3.3), gives
    }{
        在配备 Skorokhod 拓扑的 càdlàg 路径空间 $D([0,1])$ 中, 由 Bolthausen~\mycite{Bolthausen_1976} 的定理 3.2、Iglehart~\mycite{Iglehart_1974} 的第 3 节及 Caravenna 与 Chaumont~\mycite{Caravenna_Chaumont_2008} 的公式 (3.3) 中对 Brownian 蜿蜒的识别, 得到条件不变原理
    }
    \begin{equation*}
        \mathcal L\left(\left.\left(\frac{Z_{\lfloor Nt\rfloor}^\theta}{\sigma\sqrt N}\right)_{0\le t\le1}\,\right|C_N^\theta\right)
        \Longrightarrow \mathsf m,
    \end{equation*}
    \bilingual{
        where $\mathsf m$ is the standard Brownian meander. Fix $\rho:=\frac1{100}$ and let
    }{
        其中 $\mathsf m$ 为标准 Brownian 蜿蜒. 令 $\rho:=\frac1{100}$, 并定义
    }
    \begin{equation*}
        \mathcal U:=\left\{f\in D([0,1]):
            \sigma\sup_{0\le t\le1}f(t)<\frac56-\rho,\quad
            \sigma f(1)\in\left(\frac5{12}+\rho,\frac{11}{24}-\rho\right)
        \right\}.
    \end{equation*}
    \bilingual{
        The set $\mathcal U$ is open in the Skorokhod topology, and the support of the Brownian meander contains the path $f_0(t)=\frac{7t}{16\sigma}\in\mathcal U$. Thus $q_*:=\mathsf m(\mathcal U)>0$. For each fixed integer $m\ge0$, put $N=n_\Delta-m$; Portmanteau and $\Delta/\sqrt N\to1$ yield
    }{
        $\mathcal U$ 在 Skorokhod 拓扑下是开集, Brownian 蜿蜒的支撑包含路径 $f_0(t)=\frac{7t}{16\sigma}\in\mathcal U$. 故 $q_*:=\mathsf m(\mathcal U)>0$. 对每个固定整数 $m\ge0$, 令 $N=n_\Delta-m$; 由 Portmanteau 定理及 $\Delta/\sqrt N\to1$, 得
    }
    \begin{equation*}
        \liminf_{\Delta\to\infty}
        \mathbb P\left(D_{n_\Delta-m,\Delta}^\theta\middle|C_{n_\Delta-m}^\theta\right)
        \ge q_*.
    \end{equation*}
    \bilingual{
        To bound the survival probability, write $F_N^\theta:=\mathbb P(C_N^\theta)$ and $p_N^\theta:=\mathbb P(Z_N^\theta\ge0)$. The finite coefficient form of the Sparre--Andersen identity in Bauer, Godrèche and Luck~\mycite{Bauer_Godreche_Luck_1999}, Appendix A, (A.7), gives $F_0^\theta=1$ and
    }{
        为估计存活概率, 令 $F_N^\theta:=\mathbb P(C_N^\theta)$ 与 $p_N^\theta:=\mathbb P(Z_N^\theta\ge0)$. Bauer、Godrèche 与 Luck~\mycite{Bauer_Godreche_Luck_1999} 的附录 A 公式 (A.7) 所对应的 Sparre--Andersen 恒等式的有限系数形式给出 $F_0^\theta=1$ 以及
    }
    \begin{equation*}
        NF_N^\theta=\sum_{k=0}^{N-1}F_k^\theta p_{N-k}^\theta,
        \qquad N\ge1.
    \end{equation*}
    \bilingual{
        By the central limit theorem, $p_N^\theta\to\frac12$. Choose $0<\gamma<\beta<\frac12$. For some integer $N_0\ge1$, $p_j^\theta\ge\beta$ when $j\ge N_0$; hence the preceding identity yields
    }{
        由中心极限定理, $p_N^\theta\to\frac12$. 取 $0<\gamma<\beta<\frac12$. 存在整数 $N_0\ge1$, 使得 $j\ge N_0$ 时 $p_j^\theta\ge\beta$; 因此上述等式给出
    }
    \begin{equation*}
        \beta\sum_{k=0}^{N-N_0}F_k^\theta\le NF_N^\theta,
        \qquad N\ge N_0.
    \end{equation*}
    \bilingual{
        The integral bound $\sum_{k=0}^{M}(k+1)^{\gamma-1}\ge\frac{(M+2)^\gamma-1}{\gamma}$ and $\gamma<\beta$ imply $\frac\beta\gamma((N-N_0+2)^\gamma-1)\ge(N+1)^\gamma$ for all sufficiently large $N$. Since $F_k^\theta>0$, choose $C>0$ small enough to cover the remaining finite set of indices. Strong induction in the preceding inequality now gives $F_N^\theta\ge C(N+1)^{\gamma-1}$ for every $N$. Take $\gamma=\frac1{2(1+\alpha)}$; then $2(1-\gamma)\le1+\alpha$. Since $n_\Delta-m\sim\Delta^2$ for each fixed $m$, the preceding conditional estimate yields a constant $c_{\alpha,m}>0$ such that
    }{
        由积分比较, $\sum_{k=0}^{M}(k+1)^{\gamma-1}\ge\frac{(M+2)^\gamma-1}{\gamma}$. 再由 $\gamma<\beta$, 充分大的 $N$ 满足 $\frac\beta\gamma((N-N_0+2)^\gamma-1)\ge(N+1)^\gamma$. 因 $F_k^\theta>0$, 可将 $C>0$ 取得充分小, 使余下有限个指标也满足下界. 对上述不等式作强归纳, 即得每个 $N$ 均满足 $F_N^\theta\ge C(N+1)^{\gamma-1}$. 取 $\gamma=\frac1{2(1+\alpha)}$, 则 $2(1-\gamma)\le1+\alpha$. 对每个固定的 $m$, 有 $n_\Delta-m\sim\Delta^2$; 结合上述条件概率估计, 存在 $c_{\alpha,m}>0$ 使得
    }
    \begin{equation*}
        \min_{\theta\in\{-1,1\}}
        \mathbb P\left(D_{n_\Delta-m,\Delta}^\theta\right)
        \ge c_{\alpha,m}\Delta^{-1-\alpha}
    \end{equation*}
    \bilingual{for all sufficiently large $\Delta$.}{对充分大的 $\Delta$ 成立.}

    \bilingual{
        For starting points in the middle interval, Brownian path support and compactness give
    }{
        对中间区间内的起点, Brownian 路径的支撑性质与紧致性给出
    }
    \begin{equation*}
        p_0:=\inf_{z\in[\frac1{12},\frac{11}{12}]}
        \mathbb P\left(
            \frac1{24}<z+\sigma B_t<\frac{23}{24}\ (0\le t\le1),\quad
            z+\sigma B_1\in\left(\frac5{12},\frac7{12}\right)
        \right)>0,
    \end{equation*}
    \bilingual{
        where $B$ is standard Brownian motion. Since $n_\Delta/\Delta^2\to1$ and $c/\Delta\to0$, Donsker's invariance principle and compactness of $[\frac1{12},\frac{11}{12}]$ imply
    }{
        其中 $B$ 为标准 Brownian 运动. 由于 $n_\Delta/\Delta^2\to1$ 且 $c/\Delta\to0$, 由 Donsker 不变原理及区间 $[\frac1{12},\frac{11}{12}]$ 的紧致性, 得
    }
    \begin{equation*}
        \inf_{x\in[\frac1{12}\Delta,\frac{11}{12}\Delta]}
        \mathbb P_x\left(
            c\le S_j\le\Delta\ (1\le j\le n_\Delta),\quad
            S_{n_\Delta}\in\left[\frac5{12}\Delta,\frac7{12}\Delta\right]
        \right)\ge\frac{p_0}{2}
    \end{equation*}
    \bilingual{for sufficiently large $\Delta$.}{对充分大的 $\Delta$ 成立.}

    \bilingual{
        The two edge intervals use a fixed initial segment followed by a meander segment. Since the increment law is nondegenerate and centered, choose $0<a_+<b_+$ and $0<a_-<b_-$ such that
    }{
        两侧区间均先取固定长度的初始段, 再接 Brownian 蜿蜒段. 由于增量分布非退化且均值为零, 可取 $0<a_+<b_+$ 与 $0<a_-<b_-$ 使得
    }
    \begin{equation*}
        p_+:=\mathbb P(\xi_1\in[a_+,b_+])>0,
        \qquad
        p_-:=\mathbb P(-\xi_1\in[a_-,b_-])>0.
    \end{equation*}
    \bilingual{
        Fix integers $m_+,m_-$ with $m_+a_+>c+1$ and $m_-a_->c+1$. For $x\le\frac1{12}\Delta$, impose $\xi_i\in[a_+,b_+]$ for $1\le i\le m_+$ and use $D_{n_\Delta-m_+,\Delta}^{1}$ on the independent suffix. Since $h_j\le c$ for $j\le n_\Delta$, on this event, for sufficiently large $\Delta$,
    }{
        固定整数 $m_+,m_-$ 使得 $m_+a_+>c+1$ 且 $m_-a_->c+1$. 对 $x\le\frac1{12}\Delta$, 在 $1\le i\le m_+$ 时要求 $\xi_i\in[a_+,b_+]$, 并在独立后缀上取事件 $D_{n_\Delta-m_+,\Delta}^{1}$. 充分大的 $\Delta$ 下, 该事件蕴含
    }
    \begin{equation*}
        \begin{aligned}
            h_j&<S_j\le\Delta, &&1\le j\le m_+,\\
            c&<S_{m_+}+Z_j^1\le\Delta, &&1\le j\le n_\Delta-m_+,\\
            h_{n_\Delta}+\frac13\Delta
            &\le\frac5{12}\Delta\le S_{n_\Delta}
            \le\frac{13}{24}\Delta+m_+b_+
            \le h_{n_\Delta}+\frac23\Delta.
        \end{aligned}
    \end{equation*}
    \bilingual{
        Indeed, the upper bound for the suffix is $x+m_+b_++\frac56\Delta\le\frac{11}{12}\Delta+m_+b_+\le\Delta$, while the endpoint lies in $[\frac5{12}\Delta,\frac{13}{24}\Delta+m_+b_+]$. Independence and the meander estimate therefore give
    }{
        事实上, 后缀的上界为 $x+m_+b_++\frac56\Delta\le\frac{11}{12}\Delta+m_+b_+\le\Delta$, 终点位于 $[\frac5{12}\Delta,\frac{13}{24}\Delta+m_+b_+]$. 由独立性及 Brownian 蜿蜒估计, 得
    }
    \begin{equation*}
        \inf_{0\le x\le\frac1{12}\Delta}\mathbb P_x(\mathcal A_\Delta(x))
        \ge p_+^{m_+}c_{\alpha,m_+}\Delta^{-1-\alpha}.
    \end{equation*}

    \bilingual{
        For $x\ge\frac{11}{12}\Delta$, set $R_j:=\Delta-S_j$ and $y:=\Delta-x$. Require $-\xi_i\in[a_-,b_-]$ for $1\le i\le m_-$, and use $D_{n_\Delta-m_-,\Delta}^{-1}$ on the independent suffix. Since $h_j\le c$ for $j\le n_\Delta$, on this event, for sufficiently large $\Delta$,
    }{
        对 $x\ge\frac{11}{12}\Delta$, 令 $R_j:=\Delta-S_j$ 且 $y:=\Delta-x$. 在 $1\le i\le m_-$ 时要求 $-\xi_i\in[a_-,b_-]$, 并在独立后缀上取事件 $D_{n_\Delta-m_-,\Delta}^{-1}$. 充分大的 $\Delta$ 下, 有
    }
    \begin{equation*}
        \begin{aligned}
            0&\le R_j\le\Delta-c, &&1\le j\le m_-,\\
            0&\le R_{m_-}+Z_j^{-1}\le\Delta-c,
                &&1\le j\le n_\Delta-m_-,\\
            c&\le S_j\le\Delta, &&1\le j\le n_\Delta,\\
            h_{n_\Delta}+\frac13\Delta
            &\le\frac{11}{24}\Delta-m_-b_-
            \le S_{n_\Delta}\le\frac7{12}\Delta
            \le h_{n_\Delta}+\frac23\Delta.
        \end{aligned}
    \end{equation*}
    \bilingual{
        Here $R_{n_\Delta}\le\frac{13}{24}\Delta+m_-b_-$ gives $S_{n_\Delta}\ge\frac{11}{24}\Delta-m_-b_-$, and $R_{n_\Delta}\ge\frac5{12}\Delta$ gives $S_{n_\Delta}\le\frac7{12}\Delta$. Independence yields
    }{
        其中 $R_{n_\Delta}\le\frac{13}{24}\Delta+m_-b_-$ 给出 $S_{n_\Delta}\ge\frac{11}{24}\Delta-m_-b_-$, 而 $R_{n_\Delta}\ge\frac5{12}\Delta$ 给出 $S_{n_\Delta}\le\frac7{12}\Delta$. 由独立性得
    }
    \begin{equation*}
        \inf_{\frac{11}{12}\Delta\le x\le\Delta}\mathbb P_x(\mathcal A_\Delta(x))
        \ge p_-^{m_-}c_{\alpha,m_-}\Delta^{-1-\alpha}.
    \end{equation*}

    \bilingual{
        For the middle interval, $c\le S_j\le\Delta$ implies $h_j\le S_j\le\Delta+h_j$, because $h_j\le c$ for $j\le n_\Delta$. The three starting-point intervals cover $[0,\Delta]$; taking the minimum of the constants proves the claim.
    }{
        对中间区间, 当 $j\le n_\Delta$ 时有 $h_j\le c$, 故 $c\le S_j\le\Delta$ 蕴含 $h_j\le S_j\le\Delta+h_j$. 三个起点区间覆盖 $[0,\Delta]$; 取各常数的最小值即得所求.
    }
\end{proof}
