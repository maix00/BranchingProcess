\begin{lemma}
    \label{lemma:4.5}
    \bilingual{
        Let $v>0$, $0<d<\frac{\pi^2}2\sigma^2$, $c\ge 0$, $r_j\le s_j$ as in \Cref{lemma:4.2}. There exists $\Delta_0>0$ such that for all $\Delta\ge\Delta_0$, any $\ell\ge 1$ and $\delta>r_\ell$, we have
    }{
        设 $v>0$,~ $0<d<\frac{\pi^2}2\sigma^2$, $c\ge 0$ 和 $r_j\le s_j$ 如\Cref{lemma:4.2}. ~存在$\Delta_0>0$,~ 对所有$\Delta\ge\Delta_0$,~ 任何 $\ell\ge 1$ 和 $\delta>r_\ell$,~ 成立
    }
    {
        \begin{equation}
            \mathbb E\!\left[
                \left(
                    \sum_{|x|=\ell}
                    \!\!{\bf 1}_{
                        \{
                            r_j\le\mathcal V(x_j)\le s_j,\,
                            \forall j\in[\![1,\ell]\!],\,
                            \mathcal V(x)\le\delta
                        \}
                    }
                \right)^2
            \right]
            \le
            e^{4\Delta^v+2\delta-2d\frac{\ell-1}{\Delta^2}}
            \sum_{m=0}^{\ell-1}
            e^{d\frac m{\Delta^2}-r_m}
            +
            e^{\delta+\Delta^v-\frac d{\Delta^2}\ell}.
            \label{eq:4.6}
        \end{equation}
    }
\end{lemma}

\begin{lemma}
    \label{lemma:4.6}
    \bilingual{
        Let $c>0$. For $L\ge 1$, let $r_j:=\frac{\pi^2\sigma^2}{2c^2}L^{-2/3}j$ and $s_j:=r_j+cL^{1/3}$ for $j\ge 0$. For any $v>0$, we have
    }{
        令 $c>0$. ~对于 $L\ge 1$,~ 令 $r_j:=\frac{\pi^2\sigma^2}{2c^2}L^{-2/3}j$ 和 $s_j:=r_j+cL^{1/3}$ 对所有 $j\ge 0$ 成立. ~对于任何 $v>0$,~ 有
    }
    \begin{equation}
        \lim_{L\to\infty}
        L^{-\frac 13}
        \log\mathbb P(
            \exists |x|=L:\,
            r_j\le\mathcal V(x_j)\le s_j,\,
            \forall j\in[\![1,L]\!],\,
            \mathcal V(x)\le r_L+vL^{\frac 13}
        )
        =
        0.
        \label{eq:4.7}
    \end{equation}
    \bilingual{
        Moreover, 
    }{
        进一步,
    }
    \begin{equation}
        \lim_{L\to\infty}
        L^{-\frac 13}
        \log\mathbb E\!\left[
            \sum_{1\le |x|\le L}
            {\bf 1}_{
                \{
                    r_j\le\mathcal V(x_j)\le s_j,\,
                    \forall j\in[\![1,|x|]\!]
                \}
            }
        \right]
        =
        c.
        \label{eq:4.8}
    \end{equation}
\end{lemma}

The lower bound can be obtained by \Cref{lemma:4.3} and the upper bound can be obtained by \Cref{lemma:4.5,lemma:4.2}. Then we proceed to prove \eqref{eq:4.8}. The upper bound can be obtained by \Cref{lemma:4.2} and the lower bound can be obtained by \Cref{lemma:4.3}.

        于是 \eqref{eq:4.11} 得证. ~此外,~ 在上述证明中将 $F_x$ 替换为 $\widetilde F_x$ 就得到了
    }
    \begin{equation}
        \mathbb E\!\left[
            \sum_{
                |x|=\lfloor\log^{5/2}n\rfloor
            }
            \!\!\!
            \widetilde F_x
        \right]
        =
        n^{-b+v+o_n(1)}.
        \label{eq:4.11:tilde}
    \end{equation}

    \bilingual{
        Then we proceed to prove \eqref{eq:4.9}, equivalently
    }{
        接下来证明 \eqref{eq:4.9},~ 或者等价地
    }
    \begin{equation*}
        \lim_{n\to\infty}
        \frac{1}{\log n}
        \log\mathbb P\left(
            \sum_{
                |x|=\lfloor\log^{5/2}n\rfloor
            }
            \!\!\!
            F_x
            \ge
            1
        \right)
        =
        -b+v.
    \end{equation*}
    \bilingual{
        For the upper bound, we simply invoke the proved \eqref{eq:4.11}. Indeed,
    }{
        对于上界,~ 只需应用已证明的 \eqref{eq:4.11}. ~实际上,
    }
    \begin{align*}
        % &\quad
        \mathbb P\left(
            \sum_{
                |x|=\lfloor\log^{5/2}n\rfloor
            }
            \!\!\!
            F_x
            \ge
            1
        \right)
        % \\
        &
        \le
        \mathbb E\!\left[
            \left(
                \sum F_x
            \right)
            {\bf 1}_{
                \{
                    \sum F_x
                    \ge
                    1
                \}
            }
            +
            \left(
                \sum F_x
            \right)
            {\bf 1}_{
                \{
                    \sum F_x
                    =
                    0
                \}
            }
        \right]
        \\ &
        =
        \mathbb E\!\left[
            \sum_{
                |x|=\lfloor\log^{5/2}n\rfloor
            }
            \!\!\!
            F_x
        \right].
    \end{align*}
    \bilingual{
        For the lower bound, we again utilize $\widetilde F_x$ for better bound for $F_x$ and proceed to apply the Paley--Zygmund inequality. For the second moment term in the inequality, apply \eqref{eq:4.6} in \Cref{lemma:4.5} with $\Delta=(b-v')\log n$, $r_0=r+v'\log n$, $c=0$, $\ell=\lfloor\log^{5/2}n\rfloor$, $\frac 12$ in lieu of $v$, and $\delta=r+v\log n$, and we have that for any fixed $0<d<\frac{\pi^2}2\sigma^2$, any $0<v'<v<b$, some $n_4=n_4(b,v,v',d)>0$ and all $n\ge n_4$, we calculate
    }{
        对于下界,~ 再次利用 $\widetilde F_x$ 来获得 $F_x$ 的更好界,~ 并继续应用 Paley--Zygmund 不等式.~ 对于不等式中的二阶矩项,~ 应用\Cref{lemma:4.5} 中的 \eqref{eq:4.6},~ 参数设置为 $\Delta=(b-v')\log n$,~ $r_0=r+v'\log n$, $c=0$,~ $\ell=\lfloor\log^{5/2}n\rfloor$,~ 用 $\frac 12$ 代替 $v$,~ 并且 $\delta=r+v\log n$,~ 可得对于任意固定的 $0<d<\frac{\pi^2}2\sigma^2$,~ 任意 $0<v'<v<b$,~ 存在某个 $n_4=n_4(b,v,v',d)>0$ 使得所有 $n\ge n_4$,~ 计算
    }
    \begin{align*}
        e^{d\frac{m}{\Delta^2}-r_m}
        &
        \le
        e^{-r_m}
        =
        n^{b-v'},
        \\
        e^{\Delta^{1/2}}
        &
        =
        e^{
            ((b-v')\log n)^{1/2}
        }
        =
        n^{(b-v')^{1/2}\log^{-1/2}n}
        =
        n^{o_n(1)},
        \\
        \ell
        &
        =
        n^{\frac 1{\log n}\log\lfloor\log^{5/2}n\rfloor}
        \le
        n^{\frac 1{\log n}\log\log^{11/4}n}
        =
        n^{o_n(1)},
        \\
        e^{\delta-\frac{d}{\Delta^2}\ell}
        &
        =
        e^{
            \frac 1{\log n}\left(
                -b\log n+v\log n
                -\frac{d}{((b-v')\log n)^2}
                \lfloor\log^{5/2}n\rfloor
            \right)
        }
        =
        n^{-b+v+o_n(1)},
    \end{align*}
    \bilingual{
        and hence
    }{
        因此
    }
    \begin{align*}
        \mathbb E\!\left[
            \left(
                \sum_{
                    |x|=\lfloor\log^{5/2}n\rfloor
                }
                \!\!\!
                \widetilde F_x
            \right)^2
        \right]
        &
        \le
        e^{4\Delta^{1/2}+2(\delta-\frac{d}{\Delta^2}(\ell-1))}
        \sum_{m=0}^{\ell-1}
        e^{d\frac{m}{\Delta^2}-r_m}
        +
        e^{\Delta^{1/2}+\delta-\frac{d}{\Delta^2}\ell}
        \\ &
        \le
        n^{4o_n(1)+2(-b+v+o_n(1))}
        n^{o_n(1)+b-v'}
        +
        n^{o_n(1)+(-b+v+o_n(1))}
        \\ &
        =
        n^{-b+2v-v'+o_n(1)}.
    \end{align*}
    \bilingual{
        Finally, we deduce from the Paley--Zygmund inequality, \eqref{eq:4.11:tilde} and the above inequality that for all $n\ge n_4$,
    }{
        最后,~ 由 Paley--Zygmund 不等式,~ \eqref{eq:4.11:tilde} 和上述不等式推得,~ 对所有 $n\ge n_4$,
    }
    \begin{align*}
        \mathbb P\left(
            \sum_{
                |x|=\lfloor\log^{5/2}n\rfloor
            }
            \!\!\!
            F_x
            \ge
            1
        \right)
        &
        \ge
        \mathbb P\left(
            \sum_{
                |x|=\lfloor\log^{5/2}n\rfloor
            }
            \!\!\!
            \widetilde F_x
            \ge
            1
        \right)
        \\ &
        \ge
        \frac{
            \mathbb E\!\left[
                \sum_{
                    |x|=\lfloor\log^{5/2}n\rfloor
                }
                \widetilde F_x
            \right]^2
        }{
            \mathbb E\!\left[
                \left(
                    \sum_{
                        |x|=\lfloor\log^{5/2}n\rfloor
                    }
                    \widetilde F_x
                \right)^2
            \right]
        }
        \ge
        \frac{
            (n^{-b+v+o_n(1)})^2
        }{
            n^{-b+2v-v'+o_n(1)}
        }
        =
        n^{-b+v'+o_n(1)}
        ,
    \end{align*}
    \bilingual{
        and hence
    }{
        因此
    }
    \begin{equation*}
        \liminf_{n\to\infty}
        \frac{1}{\log n}
        \log\mathbb P\left(
            \sum_{
                |x|=\lfloor\log^{5/2}n\rfloor
            }
            \!\!\!
            F_x
            \ge
            1
        \right)
        \ge
        \sup_{v'\in(0,v)}
        (-b+v')
        =
        -b+v
        ,
    \end{equation*}
    \bilingual{
        which completes the proof for \eqref{eq:4.9}.
    }{
        于是 \eqref{eq:4.9} 得证.
    }

    \bilingual{
        Finally, we prove \eqref{eq:4.10}. Denote the first exit time of $[r,0]$ by $Y$ as
    }{
        最后证明 \eqref{eq:4.10}. 记 $Y$ 首次离开 $[r,0]$ 时间为
    }
    \begin{equation*}
    % $
        \tau
        :=
        \inf\left\{
            j\ge 1: Y_j\notin[r,0]
        \right\}.
    % $
    \end{equation*}
    \bilingual{
        By the many-to-one formula, we have that for any $\ell\ge 1$,
    }{
        通过\nameref{lemma:many-to-one},~ 对于任意 $\ell\ge 1$,~ 可得
    }
    \begin{align*}
        \mathbb E\!\left[
            \sum_{|x|=\ell}
            {\bf 1}_{
                \{
                    r\le\mathcal V(x_j)\le 0,\,
                    \forall j\in[\![1,\ell]\!]
                \}
            }
        \right]
        &
        =
        \mathbb E\!\left[
            e^{Y_\ell}
            {\bf 1}_{
                \{
                    r\le Y_j\le 0,\,
                    \forall j\in[\![1,\ell]\!]
                \}
            }
        \right]
        \\ &
        \le
        e^0
        \mathbb E\!\left[
            {\bf 1}_{
                \{
                    r\le Y_j\le 0,\,
                    \forall j\in[\![1,\ell]\!]
                \}
            }
        \right]
        % \\ &
        =
        \mathbb P(\tau>\ell).
    \end{align*}
    \bilingual{
        Hence,
    }{
        因此,
    }
    \begin{align*}
        \mathbb E\!\left[
            \sum_{\ell=1}^\infty
            \sum_{|x|=\ell}
            {\bf 1}_{
                \{
                    r\le\mathcal V(x_j)\le 0,\,
                    \forall j\in[\![1,\ell]\!]
                \}
            }
        \right]
        &
        =
        \sum_{\ell=1}^\infty
        \mathbb E\!\left[
            \sum_{|x|=\ell}
            {\bf 1}_{
                \{
                    r\le\mathcal V(x_j)\le 0,\,
                    \forall j\in[\![1,\ell]\!]
                \}
            }
        \right]
        \\ &
        \le
        \sum_{\ell=1}^\infty
        \mathbb P(\tau>\ell)
        =
        \mathbb E[\tau]
        -
        \mathbb P(\tau>0)
        % \\ &
        \le
        \mathbb E[\tau],
    \end{align*}
    \bilingual{
        and the result follows from the standard property
    }{
        于是由随机游走的标准性质可得
    }
    $
        \mathbb E[\tau]
        =
        O_r(|r|)
        =
        O_n(\log n)
    $.
\end{proof}
