\small
{\color{red!70!black} Notation: $\Delta$=tube width, $L$=time horizon (cf.\,$L_\Delta$ in Lemma~3.3).\\
The right-tail proof forces specific $\Delta(L)$ scales to close the spine-walk estimates.}

\medskip
\begin{itemize}
    \item \textbf{Lemma 3.8} --- 2nd moment of tube-constrained particle count at depth $L$ ($\forall\ell\ge1$, $\forall\Delta\ge\Delta_0$); cross term controlled by \textbf{Assumption~6}.
    {\color{blue!60!black} Uses Lemma~3.6.}

    \item \textbf{Lemma 3.9} --- $\Delta\sim L^{1/3}$: $\Delta=cL^{1/3}$, so $\Delta^2\ll L$ (Mogul'ski\u{i}) and $\frac{\Delta^2}{L}\log\P$ yields the $L^{1/3}$-order limit.
    \begin{align*}
        \text{Eq.~(3.7):}&\quad
        \log\P(\exists\text{ tube survivor},\,\mathcal V\le r_L+vL^{1/3})=o(L^{1/3}).\\
        \text{Eq.~(3.8):}&\quad
        \log\E[\#\text{tube paths}]=cL^{1/3}+o(L^{1/3}).
    \end{align*}
    {\color{blue!60!black} Uses Lemmas~3.6, 3.7.}

    \item \textbf{Lemma 3.10} --- $\Delta\sim\log n$: $L=\log^{5/2}n$, $\Delta=b\log n$ ($b>v>0$), so $\Delta^2\ll L$.
    \begin{align*}
        \text{Eq.~(3.9):}&\quad
        \P(\exists\text{ survivor}:\mathcal V\le r+v\log n)=n^{v-b+o(1)}.\\
        \text{Eq.~(3.10):}&\quad
        \E[\#\text{tube paths, all depths}]=O(\log n).\\
        \text{Eq.~(3.11):}&\quad
        \E[\#\text{survivors at }L:\mathcal V\le r+v\log n]=n^{-b+v+o(1)}.
    \end{align*}
    {\color{blue!60!black} Uses Lemmas~3.4, 3.7.}
\end{itemize}
