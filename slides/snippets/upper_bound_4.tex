\small
\textbf{4. What conditions are needed?}

\begin{block}{For the restart mechanism (existence of $M$)}
    \begin{itemize}
        \item $\#\Xi\ge 1$ a.s.\ and $\mathbb{E}[\#\Xi]>1$ (supercriticality)
        \item $\#\Xi^{(M)\mathrm b}\ge 1$ a.s., and $\mathbb{E}[\#\Xi^{(M)}]=m>1$
        \item $p_0:=\mathbb{P}(\#\Xi^{(M)\mathrm b}\ge 2)>0$ — a two-child event has positive probability
        \item Such $M>0$ exists because $\mathbb{P}(\#\Xi\ge 2)=p>0$ and $\lim_{M\to\infty}\mathbb{P}(\#\Xi\ge 2,\,\Xi_2\le M)=p>0$
    \end{itemize}
\end{block}

\vspace{-2mm}
\begin{block}{For the polynomial decay $\mathbb{P}(E_N^{\mathrm{c}})\le N^{-\varepsilon}$}
    \begin{itemize}
        \item $\varepsilon_1=p_1\delta>0$: $p_1=\mathbb{P}(W\ge 1)>0$ via Kesten--Stigum ($L\log L$ holds since $\#\Xi^{(M)}\le 2$)
        \item $\varepsilon_2>0$: requires $\gamma>\frac{\delta}{\theta}\mathbb{E}[e^{\theta G_1}]+\delta+\frac{\alpha}{\log m}$
        \item $\varepsilon_3>0$: needs $\mathbb{E}[e^{\lambda\Xi_1}]<\infty$ (exponential moment of $\Xi_1$) and $c$ large enough
        \item $\varepsilon_4>0$: needs $\alpha,\delta$ small enough so $\zeta'-(c+M)\delta-\frac{M}{\log m}\alpha>-\alpha+\frac{\pi^2}{2}\sigma^2 t$
    \end{itemize}
\end{block}

\vspace{-2mm}
\begin{block}{Key structural facts}
    \begin{itemize}
        \item $\kappa$ geometric trials $\Rightarrow$ $\mathbb{P}(\kappa>K)$ decays geometrically in $K=\delta\log N$
        \item $G_j$ i.i.d.\ geometric $\Rightarrow$ exponential concentration of $\sum G_j$ via Chernoff / Markov
        \item $(e^{\lambda S_k})$ is a submartingale $\Rightarrow$ Doob's maximal inequality controls $\max\mathcal{V}^{\mathrm{b}}_\tau$
    \end{itemize}
\end{block}
