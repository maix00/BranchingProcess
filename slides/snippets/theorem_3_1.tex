% Purpose: L^2 trajectory limit at the log^3-time / log-space scale; this is the key input for the speed theorem.
\begin{theorem}\label{proposition:3.1-slide}
Fix $a\in(0,1]$ and $t>0$. If $\mathcal V^N=\mathcal V^{N|a}$ starts with $\lfloor N^a\rfloor$ particles located at $0$, and its branching law $\Xi$ satisfies Assumptions~1--8, then for any sequence $\ell_N$ satisfying $\lim_{N\to\infty}\frac{\ell_N}{\log^3N}=t$,
{
    \begin{equation*}
        \frac 1{\log N}
        \min_{|u|=\ell_N}
        \mathcal V^N(u)
        \xrightarrow{\mathrm L^2}
        -a+\frac{\pi^2}2\sigma^2t.
    \end{equation*}
}
\end{theorem}
