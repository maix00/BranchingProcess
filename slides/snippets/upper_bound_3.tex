% Unified upper bound slide 3: Why P(E_N^c) ≤ N^{-ε}.

\small
\textbf{3. Bad event bound: $\mathbb{P}(E_N^{\mathrm{c}})\le N^{-\varepsilon}$.}
\[
    \mathbb{P}(E_N^{\mathrm{c}})
    \le
    \underbrace{\mathbb{P}(\kappa>K)}_{\text{Too many restarts}}
    \;+\;
    \underbrace{\mathbb{P}(\kappa\le K,\;\tau>T)}_{\text{Restarts take too long}}.
\]

\textbf{Part 1.}
The GW process $\#\mathcal Z^{(M)}$ ($m>1$) satisfies the $L\log L$ condition
($\#\Xi^{(M)}\le 2$), so \textbf{Kesten--Stigum} gives $p_1:=\mathbb{P}(W\ge 1)>0$.
Each restart succeeds independently w.p.\ $\ge p_1$, hence
$\mathbb{P}(\kappa>K)\le(1-p_1)^{\delta\log N}\le N^{-p_1\delta}$.

\textbf{Part 2.}
$G_j\sim\mathrm{Geom}(p_0)$ i.i.d.; for $\theta\in(0,-\log(1-p_0))$,
$\mathbb{E}[e^{\theta G_1}]=\frac{p_0}{1-(1-p_0)e^\theta}<\infty$.
By Chernoff, $\mathbb{P}(\kappa\le K,\tau>T)
\le\mathbb{E}[e^{\theta G_1}]^K/e^{\theta(T-K-\ell)}
=N^{-\varepsilon_2}$,
where $\varepsilon_2:=\theta\gamma-\theta(\delta+\frac{\alpha}{\log m})
-\delta\log\mathbb{E}[e^{\theta G_1}]$.
Picking $\gamma>\delta+\frac{\alpha}{\log m}
+\frac{\delta}{\theta}\log\mathbb{E}[e^{\theta G_1}]$ guarantees $\varepsilon_2>0$.

\textbf{Contribution to the upper bound.}
Cauchy--Schwarz:
$|\mathbb{E}[\mathcal{M}_{\ell_N}^{\mathrm{b}}\mathbf{1}_{E_N^{\mathrm{c}}}]|
\le\mathbb{E}[(\mathcal{M}_{\ell_N}^{\mathrm{b}})^2]^{1/2}\mathbb{P}(E_N^{\mathrm{c}})^{1/2}
\le O(\log^3 N)\cdot N^{-\varepsilon/2}\to 0$.

\textbf{Note:} $\mathbb{E}[(\Xi_1)_+^4]<\infty$ (used for the $O(\log^3 N)$ bound) is
\textbf{removable}; only $\mathbb{E}[(\Xi_1)_+]<\infty$ suffices.
