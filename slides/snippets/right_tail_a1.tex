    \item $a=1$ (start with $N$ particles at $0$ at time $0$) follows from $a\in(0,1)$ immediately.

    \[
        \P\!\left(\mathcal M^{N|1}_{\ell_N}\ge\zeta'\log N\right)
        \le
        \P\!\left(\mathcal M^{N|1/2}_{\ell_N}\ge\zeta'\log N\right)
        \le N^{-\varepsilon}.
    \]
\end{itemize}
