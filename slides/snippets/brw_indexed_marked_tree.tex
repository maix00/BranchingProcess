% Indexed marked tree families, selection mechanism, N-BRW.
\begin{itemize}
    \item \textbf{Indexed tree family:} $\boldsymbol T\in\mathbb T^{\mathbb I}$ ($\mathbb I$ = index set),
    $\sigma$-algebra $\mathcal G_{\mathbb T}^{\otimes\mathbb I}$ (product cylinder sets).
    \item \textbf{Indexed marked tree family:} $(\boldsymbol T,V)$ with domain
    $\{(u,i):u\in\boldsymbol T(i),\,i\in\mathbb I\}$, $V(\emptyset,i)=0$.
    Space: $\mathbb M^{\mathbb I}$.
    $\sigma$-algebra $\mathcal G_{\mathbb M}^{\otimes\mathbb I}$ generated by
    $\{u\in\boldsymbol T(i),\,V(u,i)\in A\}$, $u\in\mathbb V$, $i\in\mathbb I$, $A\in\mathcal B(\R)$.
    Random marked tree family: $(\boldsymbol{\mathcal T},\mathcal V)\in(\mathbb M^{\mathbb I},\mathcal G_{\mathbb M}^{\otimes\mathbb I})$.
    \item \textbf{Selection mechanism} (keep $N$ leftmost):
    a \emph{deterministic} functional $\Phi_N:\mathbb M^{\mathbb I}\to\mathbb M^{\mathbb I}$ that, at each generation, retains
    only the $N$ leftmost particles and discards the rest.
    \item \textbf{$N$-BRW:} $\mathcal V^N:=\Phi_N(\mathcal V)$ —
    the unconstrained BRW composed with the $N$-selection functional.
\end{itemize}
