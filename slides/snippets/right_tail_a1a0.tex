\begin{itemize}
    \item \textbf{$a=1$} (start with $N$ particles at $0$ at time $0$): follows from $a\in(0,1)$ immediately by stochastic dominance --- fewer particles $\Rightarrow$ larger minimum position:
    \[
        \P\!\left(\mathcal M^{N|1}_{\ell_N}\ge\zeta'\log N\right)
        \le
        \P\!\left(\mathcal M^{N|1/2}_{\ell_N}\ge\zeta'\log N\right)
        \le N^{-\varepsilon}.
    \]
    \vspace{2mm}
    \item \textbf{$a=0$} (start with 1 particle at $0$ at time $0$): requires \textbf{exponential moment of leftmost step}:
    \[
        \exists\,\lambda>0,\quad \mathbb E\big[e^{\lambda\Xi_1}\big]<\infty.
    \]
    Same \textbf{restart process} as $a\in(0,1)$: build coupled killed BRW $\mathcal V^{\mathrm b}$, reach $N^\alpha$ particles by time $\tau=o_N(\log^3 N)$, then reduce to $a\in(0,1)$ case.
    Key extra ingredient: $(e^{\lambda S_k})$ is a non-negative \textbf{submartingale} w.r.t.\ $(\mathcal F_{\tau_k})$, with Doob's maximal inequality.
\end{itemize}
