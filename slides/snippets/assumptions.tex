% Purpose: these are the standing conditions under which the main theorems are proved.
\begin{assumption}[Supercritical Regime]\label{assumption:psi-zero-slide}
    $\psi(0)>0$. That is,~ $\mathbb E[\#\Xi]>1$.
\end{assumption}
\begin{assumption}[At Least One Child]\label{assumption:at-least-one-child-slide}
    $\#\Xi\ge 1$ a.s. Or equivalently,~ $\mathbb P(\#\Xi=0)=0$.
\end{assumption}
\begin{assumption}[Exponential Moment]\label{assumption:psi-one-slide}
    $\psi(1)=0$. That is,~ $\mathbb E[W_1]=\mathbb E\left[\sum_{x\in\Xi}e^{-x}\right]=1$.
\end{assumption}
\begin{assumption}[Zero Mean]\label{assumption:psi-prime-one-slide}
    $\psi'(1)=0$. That is,~ for the spine random walk step $\xi$,~ the mean $\mathbb E[\xi]=0$.
\end{assumption}
\begin{assumption}[Finite Variance]\label{assumption:finite-variance-slide}
    $\psi''(1)<\infty$. That is,~ for the step $\xi$,~ the variance $\mathbb E[\xi^2]=\sigma^2<\infty$.
\end{assumption}
\begin{assumption}[Finite Cross Log-Laplacian]\label{assumption:log-expt-cross-sum-exp-neg-sum-slide}
    {
        \setlength\belowdisplayskip{0pt}
        \begin{equation*}
            \log\mathbb E\!\left[
                \sum_{
                    i,j\in\mathbb N,\,
                    i\neq j
                }
                e^{
                    -(\Xi_i+\Xi_j)
                }
            \right]
            <
            \infty.
        \end{equation*}
    }
\end{assumption}
\begin{assumption}[First Moment of Positive Part of Leftmost Step]\label{assumption:moment-1-slide}
    Under Assumption~\ref{assumption:at-least-one-child-slide}, we further assume that
    $\mathbb E[(\Xi_1)_+]<\infty$.
\end{assumption}
\begin{assumption}[Fourth Moment of Positive Part of Leftmost Step]\label{assumption:moment-4-slide}
    Under Assumption~\ref{assumption:moment-1-slide}, we further assume that
    $\mathbb E[(\Xi_1)_+^4]<\infty$.
\end{assumption}
