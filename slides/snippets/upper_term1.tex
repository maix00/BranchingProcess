\[
    \mathbb E\!\left[
        \left(
            \mathcal M_{\ell_N}^{\mathrm b}
            -
            \max\mathcal V^{\mathrm b}_\tau
        \right)\mathbf 1_{E_N}
    \right]
    =
    \mathbb E\!\left[
        \left.\mathbb E\!\left[
            \mathcal M^{N|\alpha}_{\ell_N-\iota}
        \right]\right|_{\iota=\tau}\mathbf 1_{E_N}
    \right]
    \le
    \left(-\alpha+\frac{\pi^2}{2}\sigma^2 t\right)\log N+o_N(\log N),
\]
where the key input is \texttt{proposition:3.1} (the $\mathrm L^2$ limit theorem) for $a=\alpha$.
