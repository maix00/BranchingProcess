\small
\begin{itemize}
    \item \textbf{Lemma 3.4} --- Horizontal tube: $\forall\Delta\ge\Delta_0$, $\forall {\color{red}L\ge1}$,
    $\P(Y_j\in[r_0,r_0{+}\Delta],\ \forall j\le L)\le e^{\Delta^v-dL/\Delta^2}$.
    \par\smallskip
    {\footnotesize\color{blue!60!black}Uses: Mogul'ski\u{i} (Lemma~3.3).}

    \item \textbf{Lemma 3.5} --- No large drop: $\forall\Delta\ge\Delta_0$, $\forall {\color{red}L\ge\Delta^{2+c}}$,
    $\P(Y_L\le\frac{\Delta}{c},\,\min\Delta Y\ge-\Delta)\le e^{-dL/\Delta^2}$.
    \par\smallskip
    {\footnotesize\color{blue!60!black}Uses: Lemma~3.4.}

    \item \textbf{Lemma 3.6} --- Tilted tube upper bound: $\forall\Delta\ge\Delta_0$, $\forall {\color{red}L\ge1}$,
    $\P(Y_j\in[r_j,s_j],\ \forall j\le L)\le e^{\Delta^v-dL/\Delta^2}$, where $r_j=r_0+\frac{c}{\Delta^2}j$, $s_j=r_j+\Delta$.
    \par\smallskip
    {\footnotesize\color{blue!60!black}Uses: Lemma~3.5.}

    \item \textbf{Lemma 3.7} --- Lower bound under tube-end constraint: $\forall\Delta\ge\Delta_0$, $\forall {\color{red}L\ge\lfloor\Delta^{2+\varepsilon}\rfloor}$,
    $\P(\text{tube to }L,\ Y_L\in[r_L,\,r_L{+}v\Delta])\ge e^{-d'L/\Delta^2}$.
    \par\smallskip
    {\footnotesize\color{blue!60!black}Uses: Mogul'ski\u{i} (Lemma~3.3).}
\end{itemize}
