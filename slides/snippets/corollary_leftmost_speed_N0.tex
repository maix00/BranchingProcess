\begin{corollary}[\texttt{corollary:leftmost-speed:N0}]
Assume $\psi(1)=0$, $\mathbb E[(\Xi_1)_+]<\infty$, and $\psi(0)>0$.
There exists $v_N=v_N(\Xi)\in\mathbb R$ such that, with probability one and in $\mathrm L^1(\mathbb P)$, for the $N$-BRW $\mathcal V^N$ with initial population $\mathcal Z^N_0\in\mathcal C(N_0)$ and $N_0\le N$,
\[
    \lim_{n\to\infty}\frac{1}{n}\mathcal M^N_n
    =
    v_N.
\]
\end{corollary}
