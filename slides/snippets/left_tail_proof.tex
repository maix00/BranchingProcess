\small
\textbf{Goal.} For $a\in[0,1]$, $\zeta<-a+\frac{\pi^2}{2}\sigma^2t$, show
\[
    \P\bigl(\mathcal M^{N|a}_{\ell_N}\le\zeta\log N\bigr)\le N^{-\varepsilon}.
\]

\medskip
\textbf{Step 1 --- Large-drop event.}
Let $b>1$. Define $F_N:=\{\exists\text{ ancestor-descendant drop}>b\log N\}$.
\[
    \P(F_N)\le N^{-(b-1)+o_N(1)}\quad\text{(branching + many-to-one).}
\]

\medskip
\textbf{Step 2 --- No-large-drop case.}
On $F_N^c$, $Y$ stays above $-b\log N$. Apply \textbf{Lemma~3.5} with
$\Delta=b\log N$, $L=\ell_N$, $c=\frac{b}{b+\zeta+a}$:
\[
    \P(\mathcal M^{N|a}_{\ell_N}\le\zeta\log N,\ F_N^c)
    \le N^{a+\zeta-dtb^{-2}+o_N(1)}.
\]

\medskip
\textbf{Step 3 --- Optimize.}
Choose $b$ close to $1$ and $d$ close to $\frac{\pi^2}{2}\sigma^2$
(Mogul'ski\u{i}, Lemma~3.3), so $a+\zeta-dtb^{-2}<0$.
Both terms $\le N^{-\varepsilon}$. $\square$
