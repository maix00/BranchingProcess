% $N$-BRW: selection mechanism and key notation.
\begin{itemize}
    \item \textbf{Init:} $\lfloor N^a\rfloor$ particles at $0$ ($a\in[0,1]$).
    \item \textbf{Branch:} each reproduces via displacement p.p.\ $\Xi$.
    \item \textbf{Selection mechanism:} keep the $N$ leftmost particles at each generation.
    \item \textbf{Leftmost position:} $\mathcal M_n^{N|a}$ at gen.\ $n$.
    \item \textbf{Speed:}
    $v_N:=\lim_{n\to\infty}\frac{1}{n}\mathcal M_n^{N|a}$
    (exists via Kingman); \\
    $v_\infty:=\lim_{n\to\infty}\frac{1}{n}\mathcal M_n$ = speed of unconstrained BRW.
\end{itemize}
