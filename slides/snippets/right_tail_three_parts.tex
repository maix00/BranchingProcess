% Right tail bound for Theorem 1.1 (a in (0,1)): compact 3-part decomposition from contents/part-5-eng.tex.
\begin{block}{Step 1: comparison via killed BRW ($a\in(0,1)$)}
\[
    \P\!\left(\mathcal M^N_{\ell_N}\ge \zeta'\log N\right)
    \le
    \P(\tau\le \ell_N)
    +
    \P\!\left(\mathcal M^{\mathrm k}_{\ell_N}\ge \zeta'\log N\right),
    \qquad
    \tau:=\inf\{k:\#\mathcal Z^{\mathrm k}_k>N\}.
\]
\end{block}

\begin{block}{Step 2: bound $\P(\tau\le \ell_N)$ via first moments}
By union bound + Markov,
\[
    \P(\tau\le \ell_N)
    \le
    \sum_{n=1}^{\ell_N}\P(\#\mathcal Z^{\mathrm k}_n>N)
    \le
    \sum_{n=1}^{\ell_N}\frac{\E[\#\mathcal Z^{\mathrm k}_n]}{N}.
\]
The thesis estimates (by Lemma 3.10 and Lemma 3.9) give:
\[
    \E[\#\mathcal Z^{\mathrm k}_n]\le N^{a+o_N(1)}\ (1\le n\le \ell),\quad
    \E[\#\mathcal Z^{\mathrm k}_{\ell+}]=N^{a-b+v+o_N(1)},\quad
    \E[\#\mathcal Z^{\mathrm k}_n]\le N^{a-b+v+ct^{1/3}+o_N(1)}\ (\ell<n\le \ell_N),
\]
so $\P(\tau\le \ell_N)\le N^{-(1-\eta)+o_N(1)}$ with $\eta:=\max\{a,\ a-b+v+ct^{1/3}\}<1$.
\end{block}

\begin{block}{Step 3: bound $\P(\mathcal M^{\mathrm k}_{\ell_N}\ge \zeta'\log N)$ via extinction at $\ell_N+$}
By the tube construction, any alive particle at time $\ell_N$ is below $\zeta'\log N$, hence
\[
    \P(\mathcal M^{\mathrm k}_{\ell_N}\ge \zeta'\log N)
    \le
    \P(\#\mathcal Z^{\mathrm k}_{\ell_N+}=0).
\]
Split at time $\ell$ using $\delta\in(0,\tfrac12(a-b+v))$:
\[
    \P(\#\mathcal Z^{\mathrm k}_{\ell_N+}=0)
    \le
    \P(\#\mathcal Z^{\mathrm k}_{\ell+}<N^\delta)
    +
    \P(\#\mathcal Z^{\mathrm k}_{\ell_N+}=0,\ \#\mathcal Z^{\mathrm k}_{\ell+}\ge N^\delta).
\]
The thesis controls the first term by a binomial lower bound + Chernoff (using Lemma 3.10), and the second term by conditioning on $\mathscr F_\ell$ and independence of subtrees.
\end{block}
