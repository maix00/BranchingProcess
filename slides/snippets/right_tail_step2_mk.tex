% Right tail Step 2: control P(M^k_{ell_N} >= zeta' log N) via extinction at ell_N+ and a split at ell+.
\begin{block}{Step 2: bound $\P(\mathcal M^{\mathrm k}_{\ell_N}\ge \zeta'\log N)$}
In the construction, any alive particle at time $\ell_N$ stays below $\zeta'\log N$ (by the parameter choice $r_{\ell_N}+vL^{1/3}<\zeta'\log N$), hence
\[
    \P(\mathcal M^{\mathrm k}_{\ell_N}\ge \zeta'\log N)
    \le
    \P(\#\mathcal Z^{\mathrm k}_{\ell_N+}=0).
\]
\end{block}
\vspace{-1mm}
Fix $\delta\in(0,\tfrac12(a-b+v))$. Split at time $\ell$:
\[
    \P(\#\mathcal Z^{\mathrm k}_{\ell_N+}=0)
    \le
    \underbrace{\P(\#\mathcal Z^{\mathrm k}_{\ell+}<N^\delta)}_{\text{few survivors at }\ell}
    +
    \underbrace{\P(\#\mathcal Z^{\mathrm k}_{\ell_N+}=0,\ \#\mathcal Z^{\mathrm k}_{\ell+}\ge N^\delta)}_{\text{many survivors but all extinct by }\ell_N}.
\]
\vspace{-1mm}
\begin{itemize}
    \item \textbf{Few particles at $\ell+$:} stochastically lower-bound $\#\mathcal Z^{\mathrm k}_{\ell+}$ by a binomial $X\sim\operatorname{Bin}(\lfloor N^a\rfloor,p)$ with $p=N^{-b+v+o_N(1)}$ via {\color{blue!60!black}Lemma~3.10 Eq.~(3.9) (Paley--Zygmund $\Rightarrow$ Lemma~3.8 $\Rightarrow$ Assumption~6)}.  Chernoff gives $\P(\#\mathcal Z^{\mathrm k}_{\ell+}<N^\delta)\le e^{-N^{\delta+o_N(1)}}$.
\end{itemize}
