\small
\textbf{Lemma 3.3 (Mogul'ski\u{i}-cal).}
Let $(L_\Delta)_{\Delta\ge0}$ satisfy
$L_\Delta\to\infty$, $\;L_\Delta/\Delta^2\to\infty$ as $\Delta\to\infty$,
and the extra condition
$N_\Delta:=\#\{\hat\Delta:L_\Delta=L_{\hat\Delta}\}=O_\Delta(1)$. Then:
\begin{align*}
    \text{(ii)}\quad
    \lim_{\Delta\to\infty}\frac{\Delta^2}{L_\Delta}
    \log\P\bigl(-a\Delta\le Y_j\le(1-a)\Delta,\ \forall j\le L_\Delta\bigr)
    &=-\frac{\pi^2}{2}\sigma^2,\\[2mm]
    \text{(iii)}\quad
    \lim_{\Delta\to\infty}\frac{\Delta^2}{L_\Delta}
    \log\P\bigl(\text{tube to }L_\Delta,\ Y_{L_\Delta}\in[(v_1-a)\Delta,(v_2-a)\Delta]\bigr)
    &=-\frac{\pi^2}{2}\sigma^2.
\end{align*}
\par\smallskip
{\footnotesize (iii) needs $0\le v_1<1$, $v_1<v_2$, $v_1,v_2=O_\Delta(1)$, same $N_\Delta$ condition.}

\medskip
These are the exact forms invoked in Lemmas~3.4 (horizontal tube) and~3.7 (tube-end constraint).
