% Right tail Step 1b: three-phase expectation estimates.
\begin{itemize}
    \item \textbf{Phase I} ($1\le n\le \ell$): particles killed outside $[-b\log N,0]$. By Lemma~3.10 Eq.~(3.10),
    \[
        \E[\#\mathcal Z^{\mathrm k}_n]
        =\sum_{i=1}^{\lfloor N^a\rfloor}\E\Bigl[\sum_{|u|=n}\one_{\{\text{stay in }[-b\log N,0]\}}\Bigr]
        \le N^a\cdot O(\log N)=N^{a+o_N(1)}.
    \]
    \item \textbf{Bottleneck at time $\ell$}: additionally kill outside $[(-b+v)\log N]$. By Lemma~3.10 Eq.~(3.11),
    \[
        \E[\#\mathcal Z^{\mathrm k}_{\ell+}]
        =\sum_{i=1}^{\lfloor N^a\rfloor}\E\Bigl[\sum_{|u|=\ell}\one_{\{\text{tube to }\ell,\;\mathcal V(u)\le(-b+v)\log N\}}\Bigr]
        =N^{a-b+v+o_N(1)}.
    \]
    \item \textbf{Phase II} ($\ell<n\le \ell_N$): conditional on $\mathcal F_\ell$, each survivor at $y$ spawns an independent subtree in tube $y+[r_j,s_j]$. By Lemma~3.9 Eq.~(3.8) + branching property,
    \[
        \E[\#\mathcal Z^{\mathrm k}_n]
        \le e^{(c+o_L(1))L^{1/3}}\cdot\E[\#\mathcal Z^{\mathrm k}_{\ell+}]
        =N^{a-b+v+ct^{1/3}+o_N(1)}.
    \]
\end{itemize}
\vspace{-1mm}
Thus $\P(\tau\le \ell_N)\le\ell_N N^{-(1-\eta)+o_N(1)}=N^{-(1-\eta)+o_N(1)}$ with $\eta:=\max\{a,\ a-b+v+ct^{1/3}\}<1$.
